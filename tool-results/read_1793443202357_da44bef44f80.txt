    95→  /* ---------------- spawning ---------------- */
    96→  private nid() { return this.nextId++; }
    97→
    98→  /**
    99→   * Pick a police spawn far away from the player's spawn point. Used by both the
   100→   * first spawn AND reset() so a replay can never seed a cop next to the player
   101→   * (which used to cause an "instant detain" right after the grace window).
   102→   */
   103→  private safeCopSpot(): { x: number; z: number } {
   104→    const avoid = [{ x: SPAWN.x, z: SPAWN.z, r: 90 }];
   105→    for (let i = 0; i < 14; i++) {
   106→      const p = freeSpot(this.rnd, this.world, 60, 140, avoid);
   107→      if (Math.hypot(p.x - SPAWN.x, p.z - SPAWN.z) >= 90) return p;
   108→    }
   109→    // Last resort: walk a wide ring around spawn until a point survives colliders.
   110→    for (let i = 0; i < 18; i++) {
   111→      const a = (i / 18) * Math.PI * 2 + this.rnd() * 0.25;
   112→      const p = { x: SPAWN.x + Math.cos(a) * 105, z: SPAWN.z + Math.sin(a) * 105 };
   113→      resolve(p, 0.9, this.world);
   114→      if (Math.hypot(p.x - SPAWN.x, p.z - SPAWN.z) >= 90) return p;
   115→    }
   116→    return { x: SPAWN.x + 105, z: SPAWN.z };
   117→  }

   119→  private spawnAll() {
   120→    const r = this.rnd;
   121→    // votes
   122→    const bands: [number, number][] = [[16, 120], [150, 215], [235, 335]];
   123→    for (let i = 0; i < TOTAL_VOTES; i++) {
   124→      const [a, b] = bands[i < 7 ? 0 : i < 12 ? 1 : 2];
   125→      const p = freeSpot(r, this.world, a, b);
   126→      this.votes.push({ id: this.nid(), x: p.x, z: p.z, state: "world" });
   127→    }
   128→    // police — keep clear of the player's spawn so nobody gets spotted on frame one
   129→    for (let i = 0; i < POLICE_COUNT; i++) {
   130→      const p = this.safeCopSpot();
   131→      this.cops.push({ id: this.nid(), kind: "police", x: p.x, z: p.z, vx: 0, vz: 0, face: 0, mode: "patrol", tx: p.x, tz: p.z, timer: 0, phase: Math.random() * 6, name: "Police", speed: POLICE_SPEED * 0.6, alertT: 0 });
   132→    }
   133→    // civilians
   134→    for (let i = 0; i < CIVILIAN_COUNT; i++) {
   135→      const p = freeSpot(r, this.world, 8, CITY_RADIUS - 8);
   136→      this.civs.push({ id: this.nid(), kind: "civilian", x: p.x, z: p.z, vx: 0, vz: 0, face: 0, mode: "wander", tx: p.x, tz: p.z, timer: range(r, 0, 4), phase: Math.random() * 6, name: CIVIL_NAMES[i % CIVIL_NAMES.length], speed: range(r, 2.2, 3.4), alertT: 0 });
   137→    }
   138→    // zombies (graveyard)
   139→    for (let i = 0; i < ZOMBIE_COUNT; i++) {
   140→      const a = range(r, 0, Math.PI * 2), d = range(r, 4, 36);
   141→      this.zombies.push({ id: this.nid(), x: GRAVEYARD.x + Math.cos(a) * d, z: GRAVEYARD.z + Math.sin(a) * d, vx: 0, vz: 0, face: 0, phase: Math.random() * 6, touchCd: 0 });
   142→    }
   143→    // animals
   144→    let ai = 0;
   145→    for (const set of ANIMAL_SET)
   146→      for (let i = 0; i < set.count; i++) {
   147→        const p = freeSpot(r, this.world, set.zone === "farm" ? CITY_RADIUS + 10 : FARM_RADIUS + 10, set.zone === "farm" ? FARM_RADIUS - 8 : JUNGLE_RADIUS - 30);
   148→        this.animals.push({ id: this.nid(), kind: set.kind, x: p.x, z: p.z, vx: 0, vz: 0, face: 0, tx: p.x, tz: p.z, timer: range(r, 0, 5), phase: Math.random() * 6, resting: false });
   149→        ai++;
   150→      }
   151→    // items
   152→    for (let i = 0; i < ITEM_COUNT; i++) {
   153→      const p = freeSpot(r, this.world, 10, 340);
   154→      this.items.push({ id: this.nid(), def: Math.floor(r() * ITEMS.length), x: p.x, z: p.z, active: true, respawn: 0 });
   155→    }
   156→  }

   170→    for (const v of this.votes) { v.state = "world"; const p = freeSpot(this.rnd, this.world, 16, 330); v.x = p.x; v.z = p.z; }
   171→    for (const it of this.items) { it.active = true; it.respawn = 0; }
   172→    for (const c of this.civs) c.mode = "wander";
   173→    // scatter cops away from spawn & calm them down so restarts are fair

   182→  /* ---------------- interactions ---------------- */
   183→  interact() {
   184→    // rescue a detained civilian nearby
   185→    let best: Npc | null = null, bd = RESCUE_RADIUS * RESCUE_RADIUS;
   186→    for (const c of this.civs) {
   187→      if (c.mode !== "detained") continue;
   188→      const d = dist2(this.player.x, this.player.z, c.x, c.z);
   189→      if (d < bd) { bd = d; best = c; }
   190→    }
   191→    if (best) {
   192→      best.mode = "rescued"; best.timer = 0;
   193→      this.rescues++; this.points += POINTS_RESCUE;
   194→      sfx.rescue();
   195→      this.fxQueue.push({ type: "confetti", x: best.x, y: 1.4, z: best.z, hue: Math.random() });
   196→      this.toast(`${best.name} is free! +${POINTS_RESCUE} support points ❤`, "good");
   197→    }
   198→  }

   200→  private dropVotes(x: number, z: number) {
   201→    for (const v of this.carriedVotes) {
   202→      const a = Math.random() * Math.PI * 2, d = range(this.rnd, 0.8, DROP_SCATTER);
   203→      const p = { x: x + Math.cos(a) * d, z: z + Math.sin(a) * d };
   204→      resolve(p, 0.8, this.world);
   205→      v.x = p.x; v.z = p.z; v.state = "world";
   206→    }
   207→    this.carriedVotes = [];
   208→  }

   210→  private catchPlayer(cop: Npc) {
   211→    const carried = this.carriedVotes.length;
   212→    this.dropVotes(this.player.x, this.player.z);
   213→    this.caughtCount++;
   214→    this.stars = Math.min(5, this.stars + 1);
   215→    this.starTimer = STAR_DECAY_S;
   216→    this.hp = Math.max(1, this.hp - 8);
   217→    this.hurtT = HURT_FLASH_S;
   218→    this.detainedT = DETAIN_SECONDS;
   219→    this.invulnT = 6; // grace window after release so cops can't instantly re-catch
   220→    cop.mode = "celebrate"; cop.timer = 2.5;
   221→    sfx.caught();
   222→    this.toast(carried > 0
   223→      ? `Caught! The police took back ${carried} vote${carried > 1 ? "s" : ""}… ⭐ Wanted level up!`
   224→      : "Caught for suspicious giggling! Sit in the time-out corner…", "bad");
   225→  }

   227→  private deliver() {
   228→    let n = 0;
   229→    while (this.carriedVotes.length) {
   230→      const v = this.carriedVotes.pop()!;
   231→      v.state = "delivered";
   232→      this.delivered++;
   233→      this.points += POINTS_DELIVER;
   234→      n++;
   235→    }
   236→    if (n > 0) {
   237→      sfx.deliver();
   238→      this.fxQueue.push({ type: "confetti", x: BALLOT.x, y: 1.6, z: BALLOT.z, hue: Math.random() });
   239→      const left = TOTAL_VOTES - this.delivered;
   240→      this.toast(n === 1 ? `Vote delivered! +${POINTS_DELIVER}` : `${n} votes delivered! +${POINTS_DELIVER * n}`, "good");
   241→      if (left === 0) {
   242→        this.won = true; this.wonAt = Date.now();
   243→        this.points += 500;
   244→        sfx.win();
   245→        this.toast("🏆 ALL VOTES DELIVERED! Democracy saved — you legend!", "good");
   246→      } else if (left <= 3) {
   247→        this.toast(`Only ${left} votes to go!`, "info");
   248→      }
   249→    }
   250→  }

   263→  /* ---------------- inventory (food in pockets) ---------------- */
   264→  /** Drop one unit of a pocketed item — tossed a little ahead so it doesn't bounce straight back into your pockets. */
   265→  dropItem(def: number) {
   266→    const idx = this.inv.findIndex((s) => s.def === def);
   267→    if (idx < 0) return;
   268→    const d = ITEMS[def];
   269→    const toss = 2.6 + Math.random() * 1.2; // past the 1.9 pickup radius
   270→    const p = { x: this.player.x + Math.sin(this.player.face) * toss, z: this.player.z + Math.cos(this.player.face) * toss };
   271→    resolve(p, 0.6, this.world);
   272→    this.items.push({ id: this.nid(), def, x: p.x, z: p.z, active: true, respawn: 0 });
   273→    this.inv[idx].count--;
   274→    if (this.inv[idx].count <= 0) this.inv.splice(idx, 1);
   275→    sfx.click();
   276→    this.toast(`${d.emoji} ${d.name} dropped — run over it to pocket it again.`, "info");
   277→  }

   279→  /** Eat/use a pocketed item: restores ❤ health and/or ⚡ energy. */
   280→  useItem(def: number) {
   281→    const idx = this.inv.findIndex((s) => s.def === def);
   282→    if (idx < 0 || this.inv[idx].count <= 0) return;
   283→    const d = ITEMS[def];
   284→    if (this.hp >= 100 && this.stamina >= 100) {
   285→      this.toast(`Already full! Save the ${d.name} for later.`, "info");
   286→      return;
   287→    }
   288→    this.inv[idx].count--;
   289→    if (this.inv[idx].count <= 0) this.inv.splice(idx, 1);
   290→    this.hp = clamp(this.hp + d.hp, 0, 100);
   291→    this.stamina = clamp(this.stamina + d.stamina, 0, 100);
   292→    this.player.stamina = this.stamina; // keep the movement body in sync
   293→    // record what this snack actually restored so the HUD can float +❤/+⚡ chips
   294→    this.lastGain = {
   295→      hp: Math.round(d.hp > 0 ? Math.min(d.hp, Math.max(0, 100 - (this.hp - d.hp))) : 0),
   296→      stamina: Math.round(d.stamina > 0 ? Math.min(d.stamina, Math.max(0, 100 - (this.stamina - d.stamina))) : 0),
   297→      at: Date.now(),
   298→    };
   299→    if (d.kind === "fruit" || d.kind === "drink") sfx.eat(); else sfx.rescue();
   300→    this.exhaustedAnnounced = false; // energy came back — allow the exhausted hint next time
   301→    this.toast(`${d.emoji} ${d.msg}`, "good");
   302→  }

   304→  /** Pick a nearby free patch of grass for an animal to wander to (local hops, not cross-map treks). */
   305→  private grazeSpot(an: Animal, farm: boolean): { x: number; z: number } | null {
   306→    const minR = farm ? CITY_RADIUS + 6 : FARM_RADIUS + 6;
   307→    const maxR = farm ? FARM_RADIUS - 10 : JUNGLE_RADIUS - 30;
   308→    for (let i = 0; i < 12; i++) {
   309→      const a = this.rnd() * Math.PI * 2, d = 7 + this.rnd() * 16;
   310→      const x = an.x + Math.cos(a) * d, z = an.z + Math.sin(a) * d;
   311→      const rr = Math.hypot(x, z);
   312→      if (rr < minR || rr > maxR) continue;
   313→      const chk = { x, z };
   314→      resolve(chk, 0.6, this.world);
   315→      if (Math.abs(chk.x - x) > 0.05 || Math.abs(chk.z - z) > 0.05) continue;
   316→      return chk;
   317→    }
   318→    return null;
   319→  }

   321→  /* ---------------- main update ---------------- */
   322→  update(dt0: number) {
   323→    if (!this.running) return;
   324→    const dt = Math.min(0.05, dt0);
   325→    this.elapsed += dt;
   326→    this.clock = advanceClock(this.clock, dt);
   327→    const night = nightFactor(this.clock);
   328→    const p = this.player;
   329→
   330→    // day/night zone toast
   331→    const z = zoneOf(p.x, p.z);
   332→    if (z !== this.zone) { this.zone = z; this.toast(ZONE_TOAST[z], "info"); }
   333→
   334→    /* --- player --- */
   335→    this.pocketFullT = Math.max(0, this.pocketFullT - dt);
   336→    this.hurtT = Math.max(0, this.hurtT - dt);
   337→    this.flipT = Math.max(0, this.flipT - dt);
   338→    this.squashT = Math.max(0, this.squashT - dt);
   339→    if (this.detainedT > 0) {
   340→      this.detainedT -= dt;
   341→      p.vx = p.vz = 0;
   342→    } else {
   343→      this.invulnT = Math.max(0, this.invulnT - dt);
   344→      const prevY = p.y;
   345→      const prevJumps = p.jumps;
   346→      p.stamina = this.stamina; // engine stamina is the single source of truth
   347→      stepBody(p, { f: this.input.f, r: this.input.r, sprint: this.input.sprint, jump: this.input.jump }, dt, this.look.age, this.world, { yaw: this.input.yaw });
   348→      this.stamina = p.stamina; // sprinting drains / walking regenerates (see stepBody)
   349→      if (p.jumps === 1 && prevJumps === 0) {
   350→        sfx.jump();
   351→        this.stamina = Math.max(0, this.stamina - JUMP_COST);
   352→      } else if (p.jumps === 2 && prevJumps === 1) {
   353→        sfx.jump2();
   354→        this.stamina = Math.max(0, this.stamina - JUMP2_COST);
   355→        this.flipT = FLIP_S;
   356→      }
   357→      if (prevY > 0 && p.y === 0) { sfx.land(); this.squashT = LAND_SQUASH_S; }
   358→      if (this.stamina <= 0.5 && !this.exhaustedAnnounced) {
   359→        this.exhaustedAnnounced = true;
   360→        this.toast("😮‍💨 Exhausted! Walk a little to catch your breath, or eat something.", "info");
   361→      } else if (this.stamina > 25) this.exhaustedAnnounced = false;
   362→    }
   363→    this.hp = Math.min(100, this.hp + 1.2 * dt * AGE_STATS[this.look.age].regen);
   364→
   365→    // wanted star decay
   366→    if (this.stars > 0) {
   367→      this.starTimer -= dt;
   368→      if (this.starTimer <= 0) { this.stars--; this.starTimer = STAR_DECAY_S; if (this.stars === 0) this.toast("Wanted level cleared. Nice and sneaky!", "info"); }
   369→    }
   370→
   371→    /* --- votes --- */
   372→    if (this.carriedVotes.length < MAX_CARRY && !this.won) {
   373→      for (const v of this.votes) {
   374→        if (v.state !== "world") continue;
   375→        if (dist2(p.x, p.z, v.x, v.z) < CARRY_RADIUS * CARRY_RADIUS) {
   376→          v.state = "carried";
   377→          this.carriedVotes.push(v);
   378→          this.points += POINTS_PICKUP;
   379→          sfx.pickup();
   380→          this.toast(`Vote snatched! (${this.carriedVotes.length}/${MAX_CARRY} carried)`, "good");
   381→          break;
   382→        }
   383→      }
   384→    }
   385→    if (this.carriedVotes.length > 0 && dist2(p.x, p.z, BALLOT.x, BALLOT.z) < DELIVER_RADIUS * DELIVER_RADIUS) this.deliver();
   386→
   387→    /* --- items: walk over food to pocket it, use it later from the HUD --- */
   388→    for (const it of this.items) {
   389→      if (!it.active) {
   390→        it.respawn -= dt;
   391→        if (it.respawn <= 0) it.active = true;
   392→        continue;
   393→      }
   394→      if (dist2(p.x, p.z, it.x, it.z) < PICKUP_RADIUS * PICKUP_RADIUS) {
   395→        const d = ITEMS[it.def];
   396→        let slot = this.inv.find((s) => s.def === it.def);
   397→        if (slot) {
   398→          if (slot.count >= INV_STACK) continue; // stack full — leave it on the ground
   399→          slot.count++;
   400→        } else {
   401→          if (this.inv.length >= INV_SLOTS) {
   402→            if (this.pocketFullT <= 0) {
   403→              this.pocketFullT = 6;
   404→              this.toast("🎒 Pockets full! Eat something to make room (press 1-8).", "info");
   405→            }
   406→            continue; // no free slot — leave it on the ground
   407→          }
   408→          this.inv.push({ def: it.def, count: 1 });
   409→          slot = this.inv[this.inv.length - 1];
   410→        }
   411→        it.active = false; it.respawn = ITEM_RESPAWN_S;
   412→        const n = this.inv.findIndex((s) => s.def === it.def) + 1;
   413→        sfx.pickup();
   414→        this.toast(`${d.emoji} ${d.name} pocketed (x${slot.count}) — tap the bag or press ${n} to use!`, "good");
   415→      }
   416→    }
   417→
   418→    /* --- police --- */
   419→    const carrying = this.carriedVotes.length;
   420→    const detect = DETECT_BASE + carrying * DETECT_CARRY_BONUS + this.stars * DETECT_STAR_BONUS;
   421→    for (const cop of this.cops) {
   422→      cop.phase += dt;
   423→      const d2p = dist2(cop.x, cop.z, p.x, p.z);
   424→      switch (cop.mode) {
   425→        case "patrol": {
   426→          cop.timer -= dt;
   427→          const arrived = dist2(cop.x, cop.z, cop.tx, cop.tz) < 4;
   428→          if (arrived || cop.timer <= 0) {
   429→            const target = freeSpot(this.rnd, this.world, 10, 120);
   430→            cop.tx = target.x; cop.tz = target.z; cop.timer = range(this.rnd, 8, 16);
   431→          }
   432→          stepNpc(cop, cop.tx, cop.tz, POLICE_SPEED * 0.6, dt, this.world);
   433→          if (this.invulnT <= 0 && this.detainedT <= 0 && d2p < detect * detect && lineClear(cop.x, cop.z, p.x, p.z, this.world)) {
   434→            cop.mode = "chase"; cop.alertT = ESCAPE_LOST_S; cop.timer = 0;
   435→            sfx.alert();
   436→            this.toast("🚨 Police spotted you! RUN!", "bad");
   437→          }
   438→          break;
   439→        }
   440→        case "chase": {
   441→          const spd = POLICE_SPRINT + this.stars * 0.22;
   442→          stepNpc(cop, p.x, p.z, spd, dt, this.world);
   443→          cop.alertT = d2p < detect * detect * 3 ? ESCAPE_LOST_S : cop.alertT - dt;
   444→          if (d2p < 1.9 * 1.9 && this.detainedT <= 0 && this.invulnT <= 0) { this.catchPlayer(cop); }
   445→          else if (cop.alertT <= 0) { cop.mode = "return"; cop.timer = 0; if (this.stars > 0) this.toast("You lost them! Phew…", "info"); }
   446→          break;
   447→        }
   448→        case "celebrate":
   449→          cop.timer -= dt;
   450→          if (cop.timer <= 0) { cop.mode = "return"; }
   451→          break;
   452→        case "return": {
   453→          if (dist2(cop.x, cop.z, cop.tx, cop.tz) < 4) cop.mode = "patrol";
   454→          else stepNpc(cop, cop.tx, cop.tz, POLICE_SPEED * 0.7, dt, this.world);
   455→          if (this.invulnT <= 0 && this.detainedT <= 0 && d2p < detect * 0.8 * (detect * 0.8) && lineClear(cop.x, cop.z, p.x, p.z, this.world)) {
   456→            cop.mode = "chase"; cop.alertT = ESCAPE_LOST_S; sfx.alert();
   457→          }
   458→          break;
   459→        }
   460→      }
   461→    }
   462→
   463→    /* --- civilians: wander + get detained + rescue --- */
   464→    for (const c of this.civs) {
   465→      c.phase += dt;
   466→      switch (c.mode) {
   467→        case "wander": {
   468→          c.timer -= dt;
   469→          if (c.timer <= 0 || dist2(c.x, c.z, c.tx, c.tz) < 2) {
   470→            const t = freeSpot(this.rnd, this.world, 6, CITY_RADIUS - 4);
   471→            c.tx = t.x; c.tz = t.z; c.timer = range(this.rnd, 6, 14);
   472→          }
   473→          stepNpc(c, c.tx, c.tz, c.speed, dt, this.world);
   474→          // random arrest event (only near player so it's participatory)
   475→          if (Math.random() < dt * 0.02 && dist2(c.x, c.z, p.x, p.z) < 70 * 70) {
   476→            c.mode = "detained"; c.timer = 35;
   477→            this.toast(`${c.name} got caught by the police! Press E near them to rescue.`, "bad");
   478→          }
   479→          break;
   480→        }
   481→        case "detained":
   482→          c.timer -= dt;
   483→          if (c.timer <= 0) { c.mode = "wander"; c.timer = 0; }
   484→          break;
   485→        case "rescued":
   486→          c.timer -= dt;
   487→          if (c.timer < 1.2) stepNpc(c, c.tx, c.tz, c.speed * 1.6, dt, this.world);
   488→          else { c.mode = "wander"; c.timer = 0; }
   489→          break;
   490→      }
   491→    }
   492→
   493→    /* --- zombies --- */
   494→    this.groanT -= dt;
   495→    for (const zb of this.zombies) {
   496→      zb.touchCd = Math.max(0, zb.touchCd - dt);
   497→      const d2p = dist2(zb.x, zb.z, p.x, p.z);
   498→      if (night > 0.55) {
   499→        const inGrave = dist2(zb.x, zb.z, GRAVEYARD.x, GRAVEYARD.z) < 55 * 55;
   500→        if (inGrave && d2p < 26 * 26) stepNpc(zb, p.x, p.z, ZOMBIE_NIGHT_SPEED, dt, this.world);
   501→        else {
   502→          zb.vx *= 0.95; zb.vz *= 0.95;
   503→          if (Math.random() < dt * 0.2) { const t = freeSpot(this.rnd, this.world, 4, 34); stepNpc(zb, GRAVEYARD.x + t.x * 0.2, GRAVEYARD.z + t.z * 0.2, ZOMBIE_SPEED * 0.4, dt, this.world); }
   504→        }
   505→        if (d2p < 1.6 * 1.6 && zb.touchCd <= 0 && this.detainedT <= 0) {
   506→          zb.touchCd = ZOMBIE_TOUCH_CD;
   507→          this.hp = Math.max(0, this.hp - ZOMBIE_DMG);
   508→          this.hurtT = HURT_FLASH_S;
   509→          sfx.zombie();
   510→          if (this.hp <= 0) this.dizzyRespawn();
   511→        }
   512→        if (this.groanT <= 0 && d2p < 20 * 20) { this.groanT = range(this.rnd, 2, 6); sfx.zombie(); }
   513→      }
   514→    }
   515→
   516→    /* --- animals --- */
   517→    for (const an of this.animals) {
   518→      an.timer -= dt;
   519→      const d2p = dist2(an.x, an.z, p.x, p.z);
   520→      const fleeR = an.kind === "cow" ? 4.5 : 6;
   521→      if (d2p < fleeR * fleeR) {
   522→        an.resting = false; // panic overrides grazing
   523→        const dx = an.x - p.x, dz = an.z - p.z, d = Math.hypot(dx, dz) || 1;
   524→        const fs = an.kind === "cow" ? 2.4 : an.kind === "chicken" ? 4.6 : an.kind === "deer" ? 6.2 : 5;
   525→        stepNpc(an, an.x + (dx / d) * 8, an.z + (dz / d) * 8, fs, dt, this.world);
   526→      } else {
   527→        const arrived = dist2(an.x, an.z, an.tx, an.tz) < 2;
   528→        if (an.timer <= 0 || (arrived && !an.resting)) {
   529→          const farm = an.kind === "cow" || an.kind === "chicken";
   530→          if (!an.resting) {
   531→            // reached the spot (or ran out of time en route) → stop and graze a while
   532→            an.resting = true;
   533→            an.tx = an.x; an.tz = an.z;
   534→            an.timer = range(this.rnd, 2.5, 6.5);
   535→          } else {
   536→            // rest over → amble to a nearby patch of grass
   537→            an.resting = false;
   538→            const spot = this.grazeSpot(an, farm);
   539→            if (spot) { an.tx = spot.x; an.tz = spot.z; }
   540→            an.timer = range(this.rnd, 5, 12);
   541→          }
   542→        }
   543→        if (!an.resting) {
   544→          stepNpc(an, an.tx, an.tz, an.kind === "cow" ? 1.1 : an.kind === "chicken" ? 1.6 : 1.8, dt, this.world);
   545→        }
   546→      }
   547→    }
   548→
   549→    // dizzy check (zombies can't be the only source)
   550→    if (this.hp <= 0 && this.detainedT <= 0) this.dizzyRespawn();
   551→
   552→    /* --- HUD snapshot --- */
   553→    this.hudT += dt;
   554→    if (this.hudT > 0.1) { this.hudT = 0; this.hud = this.buildHud(); }
   555→  }
   556→
   557→  private buildHud(): HudData {
   558→    const p = this.player;
   559→    let prompt = "";
   560→    for (const c of this.civs)
   561→      if (c.mode === "detained" && dist2(p.x, p.z, c.x, c.z) < RESCUE_RADIUS * RESCUE_RADIUS) { prompt = `Press E to rescue ${c.name}!`; break; }
   562→    if (!prompt && this.carriedVotes.length > 0 && dist2(p.x, p.z, BALLOT.x, BALLOT.z) < DELIVER_RADIUS * DELIVER_RADIUS * 1.8) prompt = "Walk closer to the ballot box to deliver!";
   563→    let policeNear = false;
   564→    for (const cop of this.cops) if (cop.mode === "chase" && dist2(cop.x, cop.z, p.x, p.z) < 30 * 30) { policeNear = true; break; }
   565→    return {
   566→      hp: Math.round(this.hp), stamina: Math.round(this.stamina), stars: this.stars,
   567→      carried: this.carriedVotes.length, delivered: this.delivered, total: TOTAL_VOTES,
   568→      points: this.points, rescues: this.rescues, support: this.rescues,
   569→      zone: this.zone, clock: this.clock, night: nightFactor(this.clock),
   570→      detained: this.detainedT > 0, detainLeft: Math.max(0, this.detainedT),
   571→      prompt, policeNear,
   572→      civilians: this.civs.length, police: this.cops.length,
   573→      won: this.won, elapsed: this.elapsed,
   574→      inv: this.inv.map((s) => ({ def: s.def, count: s.count })),
   575→      hurt: this.hurtT > 0, sprinting: p.sprinting, exhausted: this.stamina <= 0.5,
   576→      px: Math.round(this.player.x), pz: Math.round(this.player.z),
   577→      gain: this.lastGain,
   578→    };
   579→  }
   580→
   581→  getHud(): HudData { return this.hud; }
   582→
   583→  /** camera target helpers */
   584→  camTarget(): { x: number; y: number; z: number } {
   585→    const p = this.player;
   586→    return { x: p.x, y: p.y + 1.7, z: p.z };
   587→  }

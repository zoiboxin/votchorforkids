    18→    18→function Minimap({ engine }: { engine: GameEngine }) {
    19→    19→  const ref = useRef<HTMLCanvasElement>(null);
    20→    20→  const hud = useGame((s) => s.hud);
    21→    21→  useEffect(() => {
    22→    22→    const c = ref.current?.getContext("2d");
    23→    23→    if (!c) return;
    24→    24→    const S = 156, K = S / 700;
    25→    25→    const dot = (x: number, z: number, r: number, col: string, alpha = 1) => {
    26→    26→      c.globalAlpha = alpha;
    27→    27→      c.fillStyle = col;
    28→    28→      c.beginPath();
    29→    29→      c.arc(S / 2 + x * K, S / 2 + z * K, r, 0, 6.3);
    30→    30→      c.fill();
    31→    31→      c.globalAlpha = 1;
    32→    32→    };
    33→    33→    c.clearRect(0, 0, S, S);
    34→    34→    dot(0, 0, 335 * K, "#2d5a2e");
    35→    35→    dot(0, 0, 200 * K, "#7a8f3f");
    36→    36→    dot(0, 0, 130 * K, "#8d8a80");
    37→    37→    dot(-185, 165, 42 * K, "#3a3a46");
    38→    38→    // parliament
    39→    39→    c.fillStyle = "#e0483a";
    40→    40→    c.fillRect(S / 2 - 4, S / 2 - 118 * K - 3, 8, 6);
    41→    41→    // ballot
    42→    42→    c.fillStyle = "#f7c948";
    43→    43→    c.beginPath(); c.arc(S / 2, S / 2 - 100 * K, 2.6, 0, 6.3); c.fill();
    44→    44→    for (const it of engine.items) if (it.active) dot(it.x, it.z, 1.1, "#ffd166", 0.8);
    45→    45→    for (const v of engine.votes) if (v.state === "world") dot(v.x, v.z, 2.2, "#f7c948");
    46→    46→    for (const cop of engine.cops) {
    47→    47→      const chasing = cop.mode === "chase";
    48→    48→      dot(cop.x, cop.z, chasing ? 3 : 2, chasing ? "#ff4d4d" : "#6aa0ff");
    49→    49→    }
    50→    50→    for (const cv of engine.civs) if (cv.mode === "detained") dot(cv.x, cv.z, 3, "#ff8fa3");
    51→    51→    dot(engine.player.x, engine.player.z, 3.6, "#ffffff");
    52→    52→  }, [engine, hud]);
    53→    53→  return (
    54→    54→    <div className="relative origin-top-right scale-[0.72] sm:scale-100">
    55→    55→      <canvas
    56→    56→        ref={ref} width={156} height={156}
    57→    57→        aria-label="Minimap, north at the top"
    58→    58→        className="rounded-xl border-2 border-white/30 shadow-lg"
    59→    59→      />
    60→    60→      <span className="absolute -top-1.5 left-1/2 -translate-x-1/2 rounded-full border border-white/30 bg-black/70 px-1.5 text-[9px] font-black leading-3 text-white">N</span>
    61→    61→    </div>
    62→    62→  );
    63→    63→}

   152→   152→/* ------------- danger compass: arrow toward nearest chasing cop ------------- */
   153→   153→function DangerCompass({ engine }: { engine: GameEngine }) {
   154→   154→  const hud = useGame((s) => s.hud);
   155→   155→  if (!hud.policeNear || hud.detained) return null;
   156→   156→  let best: { d: number; a: number } | null = null;
   157→   157→  for (const cop of engine.cops) {
   158→   158→    if (cop.mode !== "chase") continue;
   159→   159→    const dx = cop.x - engine.player.x, dz = cop.z - engine.player.z;
   160→   160→    const d = Math.hypot(dx, dz);
   161→   161→    if (!best || d < best.d) best = { d, a: Math.atan2(dx, dz) };
   162→   162→  }
   163→   163→  if (!best) return null;
   164→   164→  // 0° = straight ahead of the camera; positive = clockwise (to the right)
   165→   165→  const deg = ((best.a - engine.input.yaw) * 180) / Math.PI;
   166→   166→  return (
   167→   167→    <motion.div
   168→   168→      initial={{ opacity: 0, scale: 0.6 }} animate={{ opacity: 1, scale: 1 }} exit={{ opacity: 0, scale: 0.6 }}
   169→   169→      className="pointer-events-none absolute left-1/2 top-1/2 z-[5]"
   170→   170→      aria-label={`Police ${Math.round(best.d)} meters away`}
   171→   171→    >
   172→   172→      <div style={{ transform: `rotate(${deg}deg)` }}>
   173→   173→        <div
   174→   174→          className="flex flex-col items-center"
   175→   175→          style={{ transform: `translate(-50%, -128px)` }}
   176→   176→        >
   177→   177→          <motion.span animate={{ y: [0, 4, 0] }} transition={{ repeat: Infinity, duration: 0.8 }}>
   178→   178→            <ChevronUp size={26} className="text-rose-400 drop-shadow-[0_1px_4px_rgba(0,0,0,0.8)]" strokeWidth={3.5} />
   179→   179→          </motion.span>
   180→   180→          <span className="rounded-full bg-rose-600/85 px-1.5 text-[9px] font-black leading-4 text-white shadow">
   181→   181→            🚓 {Math.round(best.d)}m
   182→   182→          </span>

   189→   189→/* ----------------------------- FPS counter ----------------------------- */
   190→   190→function FpsCounter() {
   191→   191→  const [fps, setFps] = useState(0);
   192→   192→  useEffect(() => {
   193→   193→    let frames = 0, last = performance.now(), raf = 0;
   194→   194→    const tick = () => {
   195→   195→      frames++;
   196→   196→      const now = performance.now();
   197→   197→      if (now - last >= 500) { setFps(Math.round((frames * 1000) / (now - last))); frames = 0; last = now; }
   198→   198→      raf = requestAnimationFrame(tick);
   199→   199→    };
   200→   200→    raf = requestAnimationFrame(tick);
   201→   201→    return () => cancelAnimationFrame(raf);
   202→   202→  }, []);
   203→   203→  return (
   204→   204→    <div aria-label="Frames per second" className="rounded-lg bg-black/40 px-2 py-0.5 text-[10px] font-black tabular-nums text-emerald-300 backdrop-blur">
   205→   205→      {fps} FPS
   206→   206→    </div>
   207→   207→  );
   208→   208→}

   211→   211→function Joystick({ engine }: { engine: GameEngine }) {
   212→   212→  const [knob, setKnob] = useState({ x: 0, y: 0 });
   213→   213→  const ref = useRef<HTMLDivElement>(null);
   214→   214→  const activeId = useRef<number | null>(null);
   215→   215→
   216→   216→  const move = (cx: number, cy: number) => {
   217→   217→    const el = ref.current;
   218→   218→    if (!el) return;
   219→   219→    const r = el.getBoundingClientRect();
   220→   220→    const dx = Math.max(-1, Math.min(1, (cx - (r.left + r.width / 2)) / (r.width / 2)));
   221→   221→    const dy = Math.max(-1, Math.min(1, (cy - (r.top + r.height / 2)) / (r.height / 2)));
   222→   222→    setKnob({ x: dx * 34, y: dy * 34 });
   223→   223→    engine.touchMove.active = true;
   224→   224→    engine.input.f = -dy;
   225→   225→    engine.input.r = dx;
   226→   226→  };
   227→   227→  const end = () => {
   228→   228→    activeId.current = null;
   229→   229→    engine.touchMove.active = false;
   230→   230→    engine.input.f = 0; engine.input.r = 0;
   231→   231→    setKnob({ x: 0, y: 0 });
   232→   232→  };
   233→   233→
   234→   234→  return (
   235→   235→    <div
   236→   236→      ref={ref} data-ui
   237→   237→      className="relative h-32 w-32 rounded-full border-2 border-white/40 bg-white/10 backdrop-blur-sm"
   238→   238→      onTouchStart={(e) => {
   239→   239→        e.preventDefault();
   240→   240→        const t = e.changedTouches[0];
   241→   241→        activeId.current = t.identifier;
   242→   242→        move(t.clientX, t.clientY);
   243→   243→      }}
   244→   244→      onTouchMove={(e) => {
   245→   245→        e.preventDefault();
   246→   246→        for (const t of Array.from(e.changedTouches))
   247→   247→          if (t.identifier === activeId.current) move(t.clientX, t.clientY);
   248→   248→      }}
   249→   249→      onTouchEnd={end}
   250→   250→      onTouchCancel={end}
   251→   251→    >
   252→   252→      <div
   253→   253→        className="absolute left-1/2 top-1/2 h-14 w-14 -translate-x-1/2 -translate-y-1/2 rounded-full border-2 border-white/60 bg-white/30"
   254→   254→        style={{ transform: `translate(calc(-50% + ${knob.x}px), calc(-50% + ${knob.y}px))` }}
   255→   255→      />
   256→   256→    </div>
   257→   257→  );
   258→   258→}

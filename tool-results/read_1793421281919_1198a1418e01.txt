   126→/** Shared wind-clock uniform + GPU sway injected into foliage materials. */
   127→const WIND = { value: 0 };
   128→function windify(mat: THREE.Material, amp: number, key: string) {
   129→  mat.onBeforeCompile = (sh) => {
   130→    sh.uniforms.uTime = WIND;
   131→    sh.uniforms.uAmp = { value: amp };
   132→    sh.vertexShader = sh.vertexShader
   133→      .replace(
   134→        "#include <common>",
   135→        "#include <common>\nuniform float uTime;\nuniform float uAmp;",
   136→      )
   137→      .replace(
   138→        "#include <begin_vertex>",
   139→        `#include <begin_vertex>
   140→        {
   141→          float ph = instanceMatrix[3][0] * 0.35 + instanceMatrix[3][2] * 0.27;
   142→          float sw = (sin(uTime * 1.4 + ph) + 0.55 * sin(uTime * 2.4 + ph * 1.7)) * uAmp;
   143→          float lift = clamp((transformed.y + 0.5) * 1.2, 0.0, 1.0);
   144→          transformed.x += sw * lift;
   145→          transformed.z += sw * lift * 0.6;
   146→        }`,
   147→      );
   148→  };
   149→  mat.customProgramCacheKey = () => key;
   150→}

   152→function Forest() {
   153→  const trunks = useRef<THREE.InstancedMesh>(null!);
   154→  const tiers = useRef<THREE.InstancedMesh>(null!);
   155→  const blobs = useRef<THREE.InstancedMesh>(null!);
   156→  const n = useMemo(() => {
   157→    let pine = 0;
   158→    for (const t of WORLD.trees) if (t.kind === 0) pine++;
   159→    return { pine, round: WORLD.trees.length - pine };
   160→  }, []);
   161→
   162→  useLayoutEffect(() => {
   163→    const r = mulberry32(20240);
   164→    // Three stacked cone tiers — base y / radius / height are × tree scale.
   165→    const TIERS = [
   166→      { y: 4.55, rad: 2.6, h: 3.4, shade: 0.93 },
   167→      { y: 6.5, rad: 1.98, h: 2.85, shade: 1.0 },
   168→      { y: 8.15, rad: 1.42, h: 2.35, shade: 1.08 },
   169→    ];
   170→    let ti = 0, bi = 0;
   171→    WORLD.trees.forEach((t, i) => {
   172→      const rot = t.x * 0.7 + t.z * 1.3;
   173→      const hm = range(r, 0.9, 1.14); // per-tree height variety
   174→      tmp.rotation.set(0, rot, 0);
   175→      tmp.scale.set(t.s, t.s, t.s);
   176→      tmp.position.set(t.x, 1.6 * t.s, t.z);
   177→      tmp.updateMatrix();
   178→      trunks.current.setMatrixAt(i, tmp.matrix);
   179→      trunks.current.setColorAt(i, col.set(TRUNK_COLS[i % TRUNK_COLS.length]));
   180→
   181→      if (t.kind === 0) {
   182→        const g = col.set(PINE_COLS[Math.floor(r() * PINE_COLS.length)]).clone();
   183→        for (const tier of TIERS) {
   184→          tmp.rotation.set(0, rot, 0);
   185→          tmp.scale.set(tier.rad * t.s, tier.h * t.s * hm, tier.rad * t.s);
   186→          tmp.position.set(t.x, tier.y * t.s * hm, t.z);
   187→          tmp.updateMatrix();
   188→          tiers.current.setMatrixAt(ti, tmp.matrix);
   189→          tiers.current.setColorAt(ti, col.copy(g).multiplyScalar(tier.shade));
   190→          ti++;
   191→        }
   192→      } else {
   193→        const g = col.set(ROUND_COLS[Math.floor(r() * ROUND_COLS.length)]).clone();
   194→        const put = (dx: number, dy: number, dz: number, rad: number, shade: number) => {
   195→          tmp.rotation.set(0, rot, 0);
   196→          tmp.scale.setScalar(rad * t.s);
   197→          tmp.position.set(t.x + dx * t.s, dy * t.s, t.z + dz * t.s);
   198→          tmp.updateMatrix();
   199→          blobs.current.setMatrixAt(bi, tmp.matrix);
   200→          blobs.current.setColorAt(bi, col.copy(g).multiplyScalar(shade));
   201→          bi++;
   202→        };
   203→        put(0, 5.1, 0, 2.5, 1);        // main puffy crown
   204→        put(0, 6.75, 0, 1.72, 1.09);   // topknot
   205→        for (let k = 0; k < 3; k++) {  // side puffs
   206→          const a = rot + (k * Math.PI * 2) / 3;
   207→          put(Math.cos(a) * 1.6, 4.35, Math.sin(a) * 1.6, 1.58, 0.9);
   208→        }
   209→      }
   210→    });
   211→    tmp.rotation.set(0, 0, 0);
   212→    tmp.scale.setScalar(1);
   213→    trunks.current.count = WORLD.trees.length;
   214→    tiers.current.count = ti;
   215→    blobs.current.count = bi;
   216→    trunks.current.instanceMatrix.needsUpdate = true;
   217→    tiers.current.instanceMatrix.needsUpdate = true;
   218→    blobs.current.instanceMatrix.needsUpdate = true;
   219→    if (trunks.current.instanceColor) trunks.current.instanceColor.needsUpdate = true;
   220→    if (tiers.current.instanceColor) tiers.current.instanceColor.needsUpdate = true;
   221→    if (blobs.current.instanceColor) blobs.current.instanceColor.needsUpdate = true;
   222→  }, [n]);
   223→
   224→  useFrame((s) => { WIND.value = s.clock.elapsedTime; });
   225→
   226→  const trunkMat = useMemo(() => new THREE.MeshLambertMaterial(), []);
   227→  const pineMat = useMemo(() => {
   228→    const m = new THREE.MeshLambertMaterial({ flatShading: true });
   229→    windify(m, 0.05, "wind-pine");
   230→    return m;
   231→  }, []);
   232→  const blobMat = useMemo(() => {
   233→    const m = new THREE.MeshLambertMaterial();
   234→    windify(m, 0.045, "wind-blob");
   235→    return m;
   236→  }, []);
   237→
   238→  return (
   239→    <>
   240→      <instancedMesh ref={trunks} args={[undefined, undefined, WORLD.trees.length]} material={trunkMat}>
   241→        <cylinderGeometry args={[0.34, 0.52, 3.2, 7]} />
   242→      </instancedMesh>
   243→      <instancedMesh ref={tiers} args={[undefined, undefined, Math.max(1, n.pine * 3)]} material={pineMat}>
   244→        <coneGeometry args={[1, 1, 8]} />
   245→      </instancedMesh>
   246→      <instancedMesh ref={blobs} args={[undefined, undefined, Math.max(1, n.round * 5)]} material={blobMat}>
   247→        <sphereGeometry args={[1, 9, 7]} />
   248→      </instancedMesh>
   249→    </>
   250→  );
   251→}

   256→function Windows({ clockRef }: { clockRef: React.MutableRefObject<number> }) {
   257→  const glass = useRef<THREE.InstancedMesh>(null!);
   258→  const glow = useRef<THREE.InstancedMesh>(null!);
   259→  const { panes, lit } = useMemo(() => {
   260→    const r = mulberry32(555);
   261→    const panes: WinInst[] = [];
   262→    const lit: WinInst[] = [];
   263→    for (const b of WORLD.buildings) {
   264→      const floors = Math.min(5, Math.max(1, Math.floor((b.h - 3) / 4)));
   265→      const cols = Math.max(1, Math.min(3, Math.floor(b.w * 2 / 5)));
   266→      for (let f = 0; f < floors; f++) {
   267→        const y = 4.2 + f * 4.1;
   268→        if (y > b.h - 2) break;
   269→        for (let c = 0; c < cols; c++) {
   270→          const off = (c - (cols - 1) / 2) * 3.1;
   271→          // front/back faces (±z)
   272→          for (const sd of [-1, 1]) {
   273→            const w: WinInst = { x: b.x + off, y, z: b.z + sd * (b.d + 0.07), ry: sd > 0 ? 0 : Math.PI };
   274→            panes.push(w);
   275→            if (r() < 0.68) lit.push(w);
   276→          }
   277→          // side faces (±x)
   278→          for (const sd of [-1, 1]) {
   279→            const w: WinInst = { x: b.x + sd * (b.w + 0.07), y, z: b.z + off, ry: sd > 0 ? Math.PI / 2 : -Math.PI / 2 };
   280→            panes.push(w);
   281→            if (r() < 0.68) lit.push(w);
   282→          }
   283→        }
   284→      }
   285→    }
   286→    return { panes, lit };
   287→  }, []);
   288→
   289→  useLayoutEffect(() => {
   290→    panes.forEach((w, i) => {
   291→      tmp.position.set(w.x, w.y, w.z);
   292→      tmp.rotation.set(0, w.ry, 0);
   293→      tmp.scale.set(1.15, 1.4, 1);
   294→      tmp.updateMatrix();
   295→      glass.current.setMatrixAt(i, tmp.matrix);
   296→    });
   297→    glass.current.instanceMatrix.needsUpdate = true;
   298→  }, [panes]);
   299→  useLayoutEffect(() => {
   300→    lit.forEach((w, i) => {
   301→      tmp.position.set(w.x, w.y, w.z);
   302→      tmp.rotation.set(0, w.ry, 0);
   303→      tmp.scale.set(1.15, 1.4, 1);
   304→      tmp.updateMatrix();
   305→      glow.current.setMatrixAt(i, tmp.matrix);
   306→    });
   307→    glow.current.count = lit.length;
   308→    glow.current.instanceMatrix.needsUpdate = true;
   309→  }, [lit]);
   310→
   311→  const glassMat = useRef<THREE.MeshLambertMaterial>(null!);
   312→  const glowMat = useRef<THREE.MeshBasicMaterial>(null!);
   313→  const dayC = useMemo(() => new THREE.Color("#5b6a7d"), []);
   314→  const nightC = useMemo(() => new THREE.Color("#1c2230"), []);
   315→  const skyC = useMemo(() => new THREE.Color(), []);
   316→  useFrame(() => {
   317→    const n = nightFactor(clockRef.current);
   318→    if (glassMat.current) glassMat.current.color.copy(skyC.copy(dayC).lerp(nightC, n));
   319→    if (glowMat.current && glow.current) {
   320→      glowMat.current.opacity = Math.max(0, n - 0.35) * 1.5 * (0.82 + Math.sin(performance.now() / 700) * 0.06);
   321→      glow.current.visible = glowMat.current.opacity > 0.02;
   322→    }
   323→  });
   324→
   325→  return (
   326→    <>
   327→      <instancedMesh ref={glass} args={[undefined, undefined, panes.length]}>
   328→        <planeGeometry />
   329→        <meshLambertMaterial ref={glassMat} color="#5b6a7d" />
   330→      </instancedMesh>
   331→      <instancedMesh ref={glow} args={[undefined, undefined, Math.max(1, lit.length)]} frustumCulled={false} visible={false}>
   332→        <planeGeometry />
   333→        <meshBasicMaterial ref={glowMat} color="#ffd76a" transparent opacity={0} depthWrite={false} />
   334→      </instancedMesh>
   335→    </>
   336→  );
   337→}

   408→function Mountains() {
   409→  const b5 = useRef<THREE.InstancedMesh>(null!);
   410→  const b6 = useRef<THREE.InstancedMesh>(null!);
   411→  const b7 = useRef<THREE.InstancedMesh>(null!);

   416→
   417→  const data = useMemo(() => {
   418→    const r = mulberry32(7);
   419→    // Major peaks — ring around the map, each with its own silhouette.
   420→    const majors: Peak[] = Array.from({ length: 26 }, (_, i) => {
   421→      const a = (i / 26) * Math.PI * 2 + range(r, -0.06, 0.06);
   422→      const d = range(r, 348, 418);
   423→      const h = range(r, 85, 178);
   424→      const w = range(r, 52, 96);
   425→      return {
   426→        x: Math.cos(a) * d, z: Math.sin(a) * d, h, w,
   427→        rot: range(r, 0, Math.PI * 2),
   428→        sx: range(r, 0.8, 1.2), sz: range(r, 0.8, 1.2),
   429→        ci: i % ROCK_COLS.length, tint: range(r, -0.045, 0.055),
   430→        seg: (i % 3) as 0 | 1 | 2,
   431→        snow: h > 125 ? range(r, 0.5, 0.64) : h > 98 ? range(r, 0.6, 0.75) : range(r, 0.78, 0.94),
   432→      };
   433→    });
   434→    // Shoulder peaks — cluster beside majors to form ranges.
   435→    const shoulders: Peak[] = [];
   436→    majors.forEach((m, i) => {
   437→      const nSh = r() < 0.45 ? 2 : r() < 0.75 ? 1 : 0;
   438→      for (let k = 0; k < nSh; k++) {
   439→        const a = Math.atan2(m.z, m.x) + (k === 0 ? 1 : -1) * range(r, 0.02, 0.055);
   440→        const d = Math.hypot(m.x, m.z) + range(r, -26, 44);
   441→        const h = m.h * range(r, 0.36, 0.6);
   442→        shoulders.push({
   443→          x: Math.cos(a) * d, z: Math.sin(a) * d, h,
   444→          w: m.w * range(r, 0.42, 0.6),
   445→          rot: range(r, 0, Math.PI * 2),
   446→          sx: range(r, 0.85, 1.15), sz: range(r, 0.85, 1.15),
   447→          ci: (i + k + 1) % ROCK_COLS.length, tint: range(r, -0.04, 0.05),
   448→          seg: ((i + k + 1) % 3) as 0 | 1 | 2,
   449→          snow: h > 75 ? range(r, 0.58, 0.72) : range(r, 0.82, 0.96),
   450→        });
   451→      }
   452→    });
   453→    // Rounded low-poly foothills, kept outside the playable jungle ring.
   454→    const hills = Array.from({ length: 44 }, (_, i) => {
   455→      const a = range(r, 0, Math.PI * 2), d = range(r, 342, 382);
   456→      return { x: Math.cos(a) * d, z: Math.sin(a) * d, h: range(r, 9, 26), w: range(r, 24, 54), ci: i % HILL_COLS.length, tint: range(r, -0.05, 0.06) };
   457→    });
   458→    const segBody = [0, 0, 0], segCap = [0, 0, 0];
   459→    for (const p of [...majors, ...shoulders]) { segBody[p.seg]++; if (p.snow < 0.98) segCap[p.seg]++; }
   460→    return { majors, shoulders, hills, segBody, segCap };
   461→  }, []);
   462→
   463→  useLayoutEffect(() => {
   464→    const bodyRefs = [b5, b6, b7], capRefs = [c5, c6, c7];
   465→    const nb = [0, 0, 0], nc = [0, 0, 0];
   466→    const putPeak = (p: Peak) => {
   467→      const br = bodyRefs[p.seg].current, cr = capRefs[p.seg].current;
   468→      tmp.rotation.set(0, p.rot, 0);
   469→      tmp.scale.set(p.w * p.sx, p.h, p.w * p.sz);
   470→      tmp.position.set(p.x, p.h / 2 - 4, p.z);
   471→      tmp.updateMatrix();
   472→      br.setMatrixAt(nb[p.seg], tmp.matrix);
   473→      br.setColorAt(nb[p.seg], col.set(ROCK_COLS[p.ci]).offsetHSL(0, 0, p.tint));
   474→      nb[p.seg]++;
   475→      if (p.snow < 0.98) {
   476→        const f = p.snow;
   477→        // Cap is the cone tip region — same triangle, slight overhang so snow drapes.
   478→        tmp.scale.set(p.w * p.sx * (1 - f) * 1.06, p.h * (1 - f) * 1.02, p.w * p.sz * (1 - f) * 1.06);
   479→        tmp.position.set(p.x, p.h * f + (p.h * (1 - f)) / 2 - 4, p.z);
   480→        tmp.updateMatrix();
   481→        cr.setMatrixAt(nc[p.seg], tmp.matrix);
   482→        nc[p.seg]++;
   483→      }
   484→    };
   485→    data.majors.forEach(putPeak);
   486→    data.shoulders.forEach(putPeak);
   487→    data.hills.forEach((hl, i) => {
   488→      tmp.rotation.set(0, hl.x * 0.11, 0);
   489→      tmp.scale.set(hl.w, hl.h, hl.w);
   490→      tmp.position.set(hl.x, hl.h * 0.55, hl.z);
   491→      tmp.updateMatrix();
   492→      hills.current.setMatrixAt(i, tmp.matrix);
   493→      hills.current.setColorAt(i, col.set(HILL_COLS[hl.ci]).offsetHSL(0, 0, hl.tint));
   494→    });
   495→    tmp.rotation.set(0, 0, 0);
   496→    tmp.scale.setScalar(1);
   497→    for (let s = 0; s < 3; s++) {
   498→      bodyRefs[s].current.count = nb[s];
   499→      capRefs[s].current.count = nc[s];
   500→      bodyRefs[s].current.instanceMatrix.needsUpdate = true;
   501→      capRefs[s].current.instanceMatrix.needsUpdate = true;
   502→      if (bodyRefs[s].current.instanceColor) bodyRefs[s].current.instanceColor.needsUpdate = true;
   503→    }
   504→    hills.current.instanceMatrix.needsUpdate = true;
   505→    if (hills.current.instanceColor) hills.current.instanceColor.needsUpdate = true;
   506→  }, [data]);
   507→
   508→  const rockMat = useMemo(() => new THREE.MeshLambertMaterial({ flatShading: true }), []);
   509→  const snowMat = useMemo(() => new THREE.MeshLambertMaterial({ color: "#f4f7fb", flatShading: true }), []);
   510→  const hillMat = useMemo(() => new THREE.MeshLambertMaterial({ flatShading: true }), []);
   511→
   512→  return (
   513→    <>
   514→      <instancedMesh ref={b5} args={[undefined, undefined, Math.max(1, data.segBody[0])]} material={rockMat} frustumCulled={false}>
   515→        <coneGeometry args={[1, 1, 5]} />
   516→      </instancedMesh>
   517→      <instancedMesh ref={b6} args={[undefined, undefined, Math.max(1, data.segBody[1])]} material={rockMat} frustumCulled={false}>
   518→        <coneGeometry args={[1, 1, 6]} />
   519→      </instancedMesh>
   520→      <instancedMesh ref={b7} args={[undefined, undefined, Math.max(1, data.segBody[2])]} material={rockMat} frustumCulled={false}>
   521→        <coneGeometry args={[1, 1, 7]} />
   522→      </instancedMesh>
   523→      <instancedMesh ref={c5} args={[undefined, undefined, Math.max(1, data.segCap[0])]} material={snowMat} frustumCulled={false}>
   524→        <coneGeometry args={[1, 1, 5]} />
   525→      </instancedMesh>
   526→      <instancedMesh ref={c6} args={[undefined, undefined, Math.max(1, data.segCap[1])]} material={snowMat} frustumCulled={false}>
   527→        <coneGeometry args={[1, 1, 6]} />
   528→      </instancedMesh>
   529→      <instancedMesh ref={c7} args={[undefined, undefined, Math.max(1, data.segCap[2])]} material={snowMat} frustumCulled={false}>
   530→        <coneGeometry args={[1, 1, 7]} />
   531→      </instancedMesh>
   532→      <instancedMesh ref={hills} args={[undefined, undefined, data.hills.length]} material={hillMat} frustumCulled={false}>
   533→        <icosahedronGeometry args={[1, 0]} />
   534→      </instancedMesh>
   535→    </>
   536→  );

   620→function Sky({ clockRef }: { clockRef: React.MutableRefObject<number> }) {
   621→  const stars = useRef<THREE.Points>(null!);
   622→  const sun = useRef<THREE.Mesh>(null!);
   623→  const moon = useRef<THREE.Mesh>(null!);
   624→  const starGeo = useMemo(() => {
   625→    const r = mulberry32(5);
   626→    const pos = new Float32Array(420 * 3);
   627→    for (let i = 0; i < 420; i++) {
   628→      const a = range(r, 0, Math.PI * 2), p = range(r, 0.12, 1.35), R = 640;
   629→      pos[i * 3] = Math.cos(a) * Math.cos(p) * R;
   630→      pos[i * 3 + 1] = Math.sin(p) * R * 0.7 + 40;
   631→      pos[i * 3 + 2] = Math.sin(a) * Math.cos(p) * R;
   632→    }
   633→    const g = new THREE.BufferGeometry();
   634→    g.setAttribute("position", new THREE.BufferAttribute(pos, 3));
   635→    return g;
   636→  }, []);
   637→  const PI2 = Math.PI * 2;
   638→  useFrame(() => {
   639→    const clock = clockRef.current;
   640→    const n = nightFactor(clock);
   641→    if (stars.current) (stars.current.material as THREE.PointsMaterial).opacity = Math.max(0, n - 0.25) * 1.6;
   642→    const a = PI2 * (clock - 0.25);
   643→    if (sun.current) sun.current.position.set(Math.cos(a) * 520, Math.sin(a) * 420, -180);
   644→    if (moon.current) moon.current.position.set(Math.cos(a + Math.PI) * 520, Math.sin(a + Math.PI) * 420, -180);
   645→  });
   646→  return (
   647→    <>
   648→      <points ref={stars} geometry={starGeo} frustumCulled={false}>
   649→        <pointsMaterial color="#ffffff" size={2.6} transparent opacity={0} sizeAttenuation={false} />
   650→      </points>
   651→      <mesh ref={sun}><sphereGeometry args={[22, 12, 10]} /><meshBasicMaterial color="#ffdd66" /></mesh>
   652→      <mesh ref={moon}><sphereGeometry args={[16, 12, 10]} /><meshBasicMaterial color="#e8ecf8" /></mesh>
   653→    </>
   654→  );
   655→}

   657→function Fireflies({ clockRef }: { clockRef: React.MutableRefObject<number> }) {
   658→  const ref = useRef<THREE.Points>(null!);
   659→  const geo = useMemo(() => {
   660→    const r = mulberry32(77);
   661→    const pos = new Float32Array(70 * 3);
   662→    for (let i = 0; i < 70; i++) {
   663→      const a = range(r, 0, Math.PI * 2), d = range(r, FARM_RADIUS + 10, JUNGLE_RADIUS - 15);
   664→      pos[i * 3] = Math.cos(a) * d;
   665→      pos[i * 3 + 1] = range(r, 0.6, 2.6);
   666→      pos[i * 3 + 2] = Math.sin(a) * d;
   667→    }
   668→    const g = new THREE.BufferGeometry();
   669→    g.setAttribute("position", new THREE.BufferAttribute(pos, 3));
   670→    return g;
   671→  }, []);
   672→  useFrame((s) => {
   673→    const n = nightFactor(clockRef.current);
   674→    const m = ref.current?.material as THREE.PointsMaterial | undefined;
   675→    if (!m) return;
   676→    m.opacity = Math.max(0, n - 0.4) * 2.2 * (0.7 + Math.sin(s.clock.elapsedTime * 3) * 0.3);
   677→    ref.current.position.y = Math.sin(s.clock.elapsedTime * 1.2) * 0.4;
   678→  });
   679→  return (
   680→    <points ref={ref} geometry={geo} frustumCulled={false}>
   681→      <pointsMaterial color="#ffe066" size={3.4} transparent opacity={0} sizeAttenuation={false} />
   682→    </points>
   683→  );

   691→function Env({ clockRef }: { clockRef: React.MutableRefObject<number> }) {
   692→  const { scene } = useThree();
   693→  const hemi = useRef<THREE.HemisphereLight>(null!);
   694→  const sun = useRef<THREE.DirectionalLight>(null!);
   695→  const sky = useMemo(() => new THREE.Color(), []);
   696→  const PI2 = Math.PI * 2;
   697→  useFrame(() => {
   698→    const clock = clockRef.current;
   699→    const n = nightFactor(clock);
   700→    const h = Math.sin(PI2 * (clock - 0.25));
   701→    const dk = Math.max(0, 1 - Math.abs(h) * 5);
   702→    sky.copy(DAY).lerp(NIGHT, n);
   703→    if (dk > 0) sky.lerp(DUSK, dk * 0.6);
   704→    scene.background = sky;
   705→    scene.fog = new THREE.Fog(sky, 70, n > 0.5 ? 270 : 470);
   706→    hemi.current.intensity = 0.35 + 0.55 * (1 - n);
   707→    sun.current.intensity = 0.25 + 0.95 * (1 - n);
   708→    sun.current.position.set(Math.cos(PI2 * (clock - 0.25)) * 200, Math.max(24, h * 200), 90);
   709→  });
   710→  return (
   711→    <>
   712→      <hemisphereLight ref={hemi} args={[0xffffff, 0x557744, 0.8]} />
   713→      <directionalLight ref={sun} args={[0xfff2d0, 1]} />
   714→    </>
   715→  );
   716→}

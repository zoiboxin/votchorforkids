    17→/* ------- shared world-map canvas painting (156px minimap + 440px big map) ------- */
    18→function drawWorldMap(c: CanvasRenderingContext2D, S: number, engine: GameEngine, night: boolean) {
    19→  const K = S / 700;          // world units → px
    20→  const U = S / 156;          // scale for the hand-tuned 156px marker sizes
    21→  const dot = (x: number, z: number, r: number, col: string, alpha = 1) => {
    22→    c.globalAlpha = alpha;
    23→    c.fillStyle = col;
    24→    c.beginPath();
    25→    c.arc(S / 2 + x * K, S / 2 + z * K, r, 0, 6.3);
    26→    c.fill();
    27→    c.globalAlpha = 1;
    28→  };
    29→  c.clearRect(0, 0, S, S);
    30→  dot(0, 0, 335 * K, "#2d5a2e");
    31→  dot(0, 0, 200 * K, "#7a8f3f");
    32→  dot(0, 0, 130 * K, "#8d8a80");
    33→  dot(-185, 165, 42 * K, "#3a3a46");
    34→  if (S > 300) {
    35→    // big map only: zone rings, graveyard fence and district labels
    36→    c.strokeStyle = "rgba(255,255,255,0.22)";
    37→    c.lineWidth = 1;
    38→    for (const r of [245, 320, 390]) { c.beginPath(); c.arc(S / 2, S / 2, r * K, 0, 6.3); c.stroke(); }
    39→    c.beginPath(); c.arc(S / 2 - 255 * K, S / 2 + 215 * K, 45 * K, 0, 6.3); c.stroke();
    40→    c.fillStyle = "rgba(255,255,255,0.55)";
    41→    c.font = `700 ${Math.round(7 * U)}px ui-sans-serif, system-ui, sans-serif`;
    42→    c.textAlign = "center";
    43→    c.fillText("CITY", S / 2, S / 2 + 90 * K);
    44→    c.fillText("FARMLAND", S / 2, S / 2 + 290 * K);
    45→    c.fillText("JUNGLE", S / 2 - 270 * K, S / 2 - 255 * K);
    46→    c.fillText("GRAVEYARD", S / 2 - 255 * K, S / 2 + 265 * K);
    47→  }
    48→  // parliament
    49→  c.fillStyle = "#e0483a";
    50→  c.fillRect(S / 2 - 4 * U, S / 2 - 118 * K - 3 * U, 8 * U, 6 * U);
    51→  // ballot
    52→  c.fillStyle = "#f7c948";
    53→  c.beginPath(); c.arc(S / 2, S / 2 - 100 * K, 2.6 * U, 0, 6.3); c.fill();
    54→  for (const it of engine.items) if (it.active) dot(it.x, it.z, 1.1 * U, "#ffd166", 0.8);
    55→  for (const v of engine.votes) if (v.state === "world") dot(v.x, v.z, 2.2 * U, "#f7c948");
    56→  for (const cop of engine.cops) {
    57→    const chasing = cop.mode === "chase";
    58→    dot(cop.x, cop.z, (chasing ? 3 : 2) * U, chasing ? "#ff4d4d" : "#6aa0ff");
    59→  }
    60→  for (const cv of engine.civs) if (cv.mode === "detained") dot(cv.x, cv.z, 3 * U, "#ff8fa3");
    61→  if (night) for (const zb of engine.zombies) dot(zb.x, zb.z, 2.4 * U, "#c084fc");
    62→  dot(engine.player.x, engine.player.z, 3.6 * U, "#ffffff");
    63→}

   261→/* ------------- danger compass: arrow toward nearest chasing cop ------------- */
   262→function DangerCompass({ engine }: { engine: GameEngine }) {
   263→  const hud = useGame((s) => s.hud);
   264→  if (!hud.policeNear || hud.detained) return null;
   265→  let best: { d: number; a: number } | null = null;
   266→  for (const cop of engine.cops) {
   267→    if (cop.mode !== "chase") continue;
   268→    const dx = cop.x - engine.player.x, dz = cop.z - engine.player.z;
   269→    const d = Math.hypot(dx, dz);
   270→    if (!best || d < best.d) best = { d, a: Math.atan2(dx, dz) };
   271→  }
   272→  if (!best) return null;
   273→  // 0° = straight ahead of the camera; positive = clockwise (to the right)
   274→  const deg = ((best.a - engine.input.yaw) * 180) / Math.PI;
   275→  return (
   276→    <motion.div
   277→      initial={{ opacity: 0, scale: 0.6 }} animate={{ opacity: 1, scale: 1 }} exit={{ opacity: 0, scale: 0.6 }}
   278→      className="pointer-events-none absolute left-1/2 top-1/2 z-[5]"
   279→      aria-label={`Police ${Math.round(best.d)} meters away`}
   280→    >
   281→      <div style={{ transform: `rotate(${deg}deg)` }}>
   282→        <div
   283→          className="flex flex-col items-center"
   284→          style={{ transform: `translate(-50%, -128px)` }}
   285→        >
   286→          <motion.span animate={{ y: [0, 4, 0] }} transition={{ repeat: Infinity, duration: 0.8 }}>
   287→            <ChevronUp size={26} className="text-rose-400 drop-shadow-[0_1px_4px_rgba(0,0,0,0.8)]" strokeWidth={3.5} />
   288→          </motion.span>
   289→          <span className="rounded-full bg-rose-600/85 px-1.5 text-[9px] font-black leading-4 text-white shadow">
   290→            🚓 {Math.round(best.d)}m
   291→          </span>
   292→        </div>
   293→      </div>
   294→    </motion.div>
   295→  );
   296→}

   298→/* ----------------------------- FPS counter ----------------------------- */
   299→function FpsCounter() {
   300→  const [fps, setFps] = useState(0);
   301→  useEffect(() => {
   302→    let frames = 0, last = performance.now(), raf = 0;
   303→    const tick = () => {
   304→      frames++;
   305→      const now = performance.now();
   306→      if (now - last >= 500) { setFps(Math.round((frames * 1000) / (now - last))); frames = 0; last = now; }
   307→      raf = requestAnimationFrame(tick);
   308→    };
   309→    raf = requestAnimationFrame(tick);
   310→    return () => cancelAnimationFrame(raf);
   311→  }, []);
   312→  return (
   313→    <div aria-label="Frames per second" className="rounded-lg bg-black/40 px-2 py-0.5 text-[10px] font-black tabular-nums text-emerald-300 backdrop-blur">
   314→      {fps} FPS
   315→    </div>
   316→  );
   317→}

   319→/* ----------------------------- mobile controls ----------------------------- */
   320→function Joystick({ engine }: { engine: GameEngine }) {
   321→  const [knob, setKnob] = useState({ x: 0, y: 0 });
   322→  const ref = useRef<HTMLDivElement>(null);
   323→  const activeId = useRef<number | null>(null);
   324→
   325→  const move = (cx: number, cy: number) => {
   326→    const el = ref.current;
   327→    if (!el) return;
   328→    const r = el.getBoundingClientRect();
   329→    const dx = Math.max(-1, Math.min(1, (cx - (r.left + r.width / 2)) / (r.width / 2)));
   330→    const dy = Math.max(-1, Math.min(1, (cy - (r.top + r.height / 2)) / (r.height / 2)));
   331→    setKnob({ x: dx * 34, y: dy * 34 });
   332→    engine.touchMove.active = true;
   333→    engine.input.f = -dy;
   334→    engine.input.r = dx;
   335→  };
   336→  const end = () => {
   337→    activeId.current = null;
   338→    engine.touchMove.active = false;
   339→    engine.input.f = 0; engine.input.r = 0;
   340→    setKnob({ x: 0, y: 0 });
   341→  };
   342→
   343→  return (
   344→    <div
   345→      ref={ref} data-ui
   346→      className="relative h-32 w-32 rounded-full border-2 border-white/40 bg-white/10 backdrop-blur-sm"
   347→      onTouchStart={(e) => {
   348→        e.preventDefault();
   349→        const t = e.changedTouches[0];
   350→        activeId.current = t.identifier;
   351→        move(t.clientX, t.clientY);
   352→      }}
   353→      onTouchMove={(e) => {
   354→        e.preventDefault();
   355→        for (const t of Array.from(e.changedTouches))
   356→          if (t.identifier === activeId.current) move(t.clientX, t.clientY);
   357→      }}
   358→      onTouchEnd={end}
   359→      onTouchCancel={end}
   360→    >
   361→      <div
   362→        className="absolute left-1/2 top-1/2 h-14 w-14 -translate-x-1/2 -translate-y-1/2 rounded-full border-2 border-white/60 bg-white/30"
   363→        style={{ transform: `translate(calc(-50% + ${knob.x}px), calc(-50% + ${knob.y}px))` }}
   364→      />
   365→    </div>
   366→  );
   367→}

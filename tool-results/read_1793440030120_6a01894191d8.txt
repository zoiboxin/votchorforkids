    14→function G<T extends THREE.BufferGeometry>(key: string, make: () => T): T {
    15→  let g = geoCache.get(key);
    16→  if (!g) geoCache.set(key, (g = make()));
    17→  return g as T;
    18→}

    20→/* --- cel-shaded gradient map (4 soft steps) shared by all toon mats --- */
    21→let gradientMap: THREE.DataTexture | null = null;
    22→function getGradient(): THREE.DataTexture {
    23→  if (!gradientMap) {
    24→    const steps = new Uint8Array([110, 160, 210, 255]);
    25→    gradientMap = new THREE.DataTexture(steps, steps.length, 1, THREE.RedFormat);
    26→    gradientMap.minFilter = THREE.NearestFilter;
    27→    gradientMap.magFilter = THREE.NearestFilter;
    28→    gradientMap.generateMipmaps = false;
    29→    gradientMap.needsUpdate = true;
    30→  }
    31→  return gradientMap;
    32→}

    45→const sphere = (r: number, w = 12, h = 10, ps = 0, pl = Math.PI * 2, ts = 0, tl = Math.PI) =>

    59→/** Collects primitives and merges them into one vertex-coloured mesh. */
    60→class Bucket {
    61→  parts: THREE.BufferGeometry[] = [];
    62→  add(
    63→    geo: THREE.BufferGeometry, color: string, unlit: boolean,
    64→    x = 0, y = 0, z = 0, rx = 0, ry = 0, rz = 0, sx = 1, sy = 1, sz = 1,
    65→  ) {
    66→    const g = geo.clone();
    67→    _m4.compose(_v.set(x, y, z), _q.setFromEuler(_e.set(rx, ry, rz)), _s.set(sx, sy, sz));
    68→    g.applyMatrix4(_m4);
    69→    const c = new THREE.Color(color);
    70→    const n = g.attributes.position.count;
    71→    const arr = new Float32Array(n * 3);
    72→    for (let i = 0; i < n; i++) { arr[i * 3] = c.r; arr[i * 3 + 1] = c.g; arr[i * 3 + 2] = c.b; }
    73→    g.setAttribute("color", new THREE.BufferAttribute(arr, 3));
    74→    for (const k of Object.keys(g.attributes)) if (k !== "position" && k !== "normal" && k !== "uv" && k !== "color") g.deleteAttribute(k);
    75→    this.parts.push(g);
    76→  }
    77→  mesh(kind: "toon" | "basic"): THREE.Mesh | null {
    78→    if (!this.parts.length) return null;
    79→    const merged = mergeGeometries(this.parts, false);
    80→    this.parts.forEach((p) => p.dispose());
    81→    this.parts = [];
    82→    if (!merged) return null;
    83→    merged.clearGroups();
    84→    const mesh = new THREE.Mesh(merged, kind === "toon" ? toonVC() : basicVC());
    85→    mesh.castShadow = false;
    86→    mesh.receiveShadow = false;
    87→    return mesh;
    88→  }
    89→}

   105→let shadowTex: THREE.CanvasTexture | null = null;
   106→function getShadowTex(): THREE.CanvasTexture {
   107→  if (shadowTex) return shadowTex;
   108→  const c = document.createElement("canvas");
   109→  c.width = c.height = 64;
   110→  const ctx = c.getContext("2d")!;
   111→  const g = ctx.createRadialGradient(32, 32, 2, 32, 32, 30);
   112→  g.addColorStop(0, "rgba(0,0,0,0.34)");
   113→  g.addColorStop(1, "rgba(0,0,0,0)");
   114→  ctx.fillStyle = g;
   115→  ctx.fillRect(0, 0, 64, 64);
   116→  shadowTex = new THREE.CanvasTexture(c);
   117→  return shadowTex;
   118→}
   119→export function blobShadow(r = 0.55): THREE.Mesh {
   120→  const m = new THREE.Mesh(
   121→    G("shadowPlane", () => new THREE.PlaneGeometry(1, 1)),
   122→    new THREE.MeshBasicMaterial({ map: getShadowTex(), transparent: true, depthWrite: false }),
   123→  );
   124→  m.rotation.x = -Math.PI / 2;
   125→  m.scale.setScalar(r * 2);
   126→  m.position.y = 0.11;
   127→  return m;
   128→}

   138→function addFace(
   139→  head: THREE.Group, skin: string, toon: Bucket, unlit: Bucket,
   140→  lids: THREE.Group[], brows: THREE.Group[], opts: { droopy?: boolean; scale?: number; freckles?: boolean },
   141→) {
   142→  const s = opts.scale ?? 1;
   143→
   144→  for (const side of [-1, 1]) {
   145→    const ex = side * 0.155 * s;
   146→    // white of the eye + iris + pupil + glints (unlit so they always pop)
   147→    unlit.add(sphere(0.105 * s, 10, 8), "#ffffff", true, ex, 0.03 * s, 0.315 * s, 0, 0, 0, 1, 1.32, 0.5);
   148→    unlit.add(sphere(0.062 * s, 8, 6), IRIS, true, ex, 0.028 * s, 0.36 * s, 0, 0, 0, 1, 1.25, 0.45);
   149→    unlit.add(sphere(0.031 * s, 6, 5), EYE_B, true, ex, 0.022 * s, 0.393 * s);
   150→    unlit.add(sphere(0.018 * s, 6, 5), GLINT, true, ex + 0.022 * s, 0.062 * s, 0.402 * s);
   151→    unlit.add(sphere(0.009 * s, 5, 4), GLINT, true, ex - 0.024 * s, -0.005 * s, 0.402 * s);
   152→
   153→    // eyelid — animated dome that blinks
   154→    const lid = new THREE.Group();
   155→    lid.position.set(ex, 0.03 * s, 0.315 * s);
   156→    const dome = new THREE.Mesh(lidGeo(), new THREE.MeshToonMaterial({ color: skin, gradientMap: getGradient() }));
   157→    dome.scale.set(1.02, 1.34, 0.62);
   158→    lid.add(dome);
   159→    lid.rotation.x = opts.droopy ? -0.35 : LID_OPEN;
   160→    head.add(lid);
   161→    lids.push(lid);
   162→
   163→    // brow — chunky capsule, sits proud of the hairline
   164→    const brow = new THREE.Group();
   165→    brow.position.set(ex, 0.19 * s, 0.385 * s);
   166→    const bGeo = capsule(0.024 * s, 0.09 * s).clone();
   167→    bGeo.rotateZ(Math.PI / 2);
   168→    const bm = new THREE.Mesh(bGeo, new THREE.MeshToonMaterial({ color: opts.droopy ? "#2a3218" : "#241a12", gradientMap: getGradient() }));
   169→    brow.add(bm);
   170→    brow.rotation.z = side * -0.08;
   171→    head.add(brow);
   172→    brows.push(brow);
   173→
   174→    // solid blush dot — soft rosy cheeks
   175→    toon.add(sphere(0.055 * s, 8, 6), "#ff9fb2", false, side * 0.25 * s, -0.12 * s, 0.285 * s, 0, 0, 0, 1.5, 0.95, 0.5);
   176→  }
   177→
   178→  // button nose
   179→  toon.add(sphere(0.045 * s, 8, 6), skin, false, 0, -0.045 * s, 0.395 * s, 0, 0, 0, 1.1, 0.9, 0.9);
   180→
   181→  // kid freckles — three tiny dots per cheek
   182→  if (opts.freckles)
   183→    for (const side of [-1, 1])
   184→      for (let i = 0; i < 3; i++)
   185→        unlit.add(sphere(0.011 * s, 5, 4), "#c08552", true, side * (0.2 + (i % 2) * 0.07) * s, (-0.075 - Math.floor(i / 2) * 0.035) * s, 0.315 * s);
   186→
   187→  // big open smile — arc + tiny tongue (unlit, always readable)
   188→  const smileGeo = torus(0.095 * s, 0.023 * s, Math.PI).clone();
   189→  smileGeo.rotateX(0.12);
   190→  smileGeo.rotateZ(Math.PI);
   191→  unlit.add(smileGeo, SMILE, true, 0, -0.145 * s, 0.36 * s);
   192→  unlit.add(sphere(0.038 * s, 8, 6), TONGUE, true, 0, -0.175 * s, 0.375 * s, 0, 0, 0, 1.1, 0.55, 0.6);
   193→
   194→  // ears
   195→  for (const side of [-1, 1])
   196→    toon.add(sphere(0.085 * s, 8, 6), skin, false, side * 0.385 * s, 0.01 * s, -0.02 * s, 0, 0, 0, 0.42, 0.85, 0.72);
   197→}

   418→function updateBlink(g: THREE.Object3D, rig: CharRig, t: number) {
   419→  const st = g.userData.blink as { next: number; t: number; amt: number } | undefined;
   420→  if (!st || rig.lids.length < 2) return;
   421→  if (st.t < 0) st.t = t;
   422→  if (t >= st.next) { st.t = t; st.next = t + 1.6 + Math.random() * 3.4; }
   423→  const p = (t - st.t) / 0.16;
   424→  st.amt = p >= 1 ? 0 : Math.sin(Math.min(1, p) * Math.PI);
   425→  const rot = LID_OPEN + (LID_CLOSED - LID_OPEN) * st.amt;
   426→  rig.lids[0].rotation.x = rot;
   427→  rig.lids[1].rotation.x = rot;
   428→}

   430→export function poseCharacter(g: THREE.Object3D, speed: number, t: number, o: PoseOpts = {}) {
   431→  const rig = rigOf(g);
   432→  if (!rig) return;
   433→  const { legs, arms, body, head } = rig;
   434→  const phase = o.phase ?? 0;
   435→  const HEAD_Y = 1.84;
   436→
   437→  updateBlink(g, rig, t);
   438→
   439→  if (rig.brows.length === 2) {
   440→    const sad = o.detained ? 0.3 : 0;
   441→    rig.brows[0].rotation.z = -0.08 + sad;
   442→    rig.brows[1].rotation.z = 0.08 - sad;
   443→    const lift = o.detained ? -0.015 : Math.min(0.03, speed * 0.004);
   444→    rig.brows[0].position.y = 0.19 + lift;
   445→    rig.brows[1].position.y = 0.19 + lift;
   446→  }
   447→
   448→  if (o.detained) {
   449→    body.position.y = -0.34 + Math.sin(t * 2.2 + phase) * 0.015;
   450→    body.rotation.x = 0.12;
   451→    body.rotation.y = 0;
   452→    body.rotation.z = 0;
   453→    head.position.y = HEAD_Y;
   454→    legs[0].rotation.x = legs[1].rotation.x = -1.4;
   455→    arms[0].rotation.x = arms[1].rotation.x = 0.3;
   456→    arms[0].rotation.z = 0.15; arms[1].rotation.z = -0.15;
   457→    head.rotation.x = 0.32;
   458→    body.scale.y = 1;
   459→    return;
   460→  }
   461→
   462→  head.rotation.x = 0;
   463→  body.rotation.x = 0;
   464→  body.rotation.z = 0;
   465→  body.rotation.y = 0;
   466→
   467→  if (o.airborne) {
   468→    // mid-air front flip after a double jump — one full rotation over FLIP_S
   469→    if (o.flip && o.flip > 0) {
   470→      const p = 1 - Math.min(1, o.flip);        // 0 → 1 through the flip
   471→      body.rotation.x = -Math.PI * 2 * p;       // full front flip
   472→      body.rotation.y = 0;
   473→      body.rotation.z = 0;
   474→      legs[0].rotation.x = -1.5; legs[1].rotation.x = -1.25; // tight tuck
   475→      arms[0].rotation.x = -1.15; arms[1].rotation.x = -1.15;
   476→      arms[0].rotation.z = 0.32; arms[1].rotation.z = -0.32;
   477→      head.position.y = HEAD_Y;
   478→      body.position.y = 0;
   479→      body.scale.y = 1.05; body.scale.x = body.scale.z = 0.96;
   480→      return;
   481→    }
   482→    const rising = (o.vy ?? 0) > 0;
   483→    if (rising) {
   484→      // launch: arms thrown up, legs trailing, body stretched
   485→      legs[0].rotation.x = -0.75; legs[1].rotation.x = 0.45;
   486→      arms[0].rotation.x = -2.6; arms[1].rotation.x = -2.4;
   487→      arms[0].rotation.z = 0.3; arms[1].rotation.z = -0.3;
   488→      body.rotation.x = -0.12;
   489→      body.scale.y = 1.07; body.scale.x = body.scale.z = 0.96;
   490→    } else {
   491→      // falling: superhero arms out, legs reaching for the ground
   492→      legs[0].rotation.x = -0.5; legs[1].rotation.x = 0.3;
   493→      arms[0].rotation.x = -0.85; arms[1].rotation.x = -0.85;
   494→      arms[0].rotation.z = 1.15; arms[1].rotation.z = -1.15;
   495→      body.rotation.x = 0.06;
   496→      body.scale.y = 0.97; body.scale.x = body.scale.z = 1.02;
   497→    }
   498→    body.rotation.y = 0.1;
   499→    body.rotation.z = 0;
   500→    head.position.y = HEAD_Y;
   501→    body.position.y = 0;
   502→    return;
   503→  }
   504→
   505→  // landing squash-and-stretch: peaks mid-way, eases back to 1
   506→  const sq = o.squash ? Math.sin(Math.min(1, o.squash) * Math.PI) * 0.16 : 0;
   507→
   508→  const sp = Math.min(1.6, speed / 4.5);
   509→  if (sp < 0.06) {
   510→    const b = Math.sin(t * 2 + phase);
   511→    body.position.y = b * 0.02 - sq * 0.35;
   512→    body.scale.y = (1 + b * 0.014) * (1 - sq);
   513→    body.scale.x = body.scale.z = 1 + sq * 0.7;
   514→    body.rotation.z = Math.sin(t * 0.7 + phase) * 0.022; // gentle weight shift
   515→    head.position.y = HEAD_Y + b * 0.012;
   516→    legs[0].rotation.x = legs[1].rotation.x = 0;
   517→    arms[0].rotation.x = b * 0.07; arms[1].rotation.x = -b * 0.07;
   518→    arms[0].rotation.z = 0.09 + b * 0.02; arms[1].rotation.z = -0.09 - b * 0.02;
   519→    head.rotation.y = Math.sin(t * 0.6 + phase) * 0.16;
   520→    head.rotation.z = Math.sin(t * 0.9 + phase) * 0.03;
   521→    return;
   522→  }
   523→
   524→  const sprinting = speed > 6.9;
   525→  const freq = t * (5.4 + speed * 1.45) + phase;
   526→  const k = Math.sin(freq) * sp * 0.98;
   527→  legs[0].rotation.x = k;
   528→  legs[1].rotation.x = -k;
   529→  const pump = sprinting ? 1.18 : 0.9;
   530→  arms[0].rotation.x = -k * pump;
   531→  arms[1].rotation.x = k * pump;
   532→  arms[0].rotation.z = 0.12;
   533→  arms[1].rotation.z = -0.12;
   534→  const bounce = Math.abs(Math.cos(freq));
   535→  body.position.y = bounce * 0.06 * sp - sq * 0.35;
   536→  body.scale.y = (1 - bounce * 0.035 * sp) * (1 - sq);
   537→  body.scale.x = body.scale.z = (1 + bounce * 0.02 * sp) * (1 + sq * 0.7);
   538→  body.rotation.x = Math.min(0.3, sp * 0.13) + (sprinting ? 0.14 : 0);
   539→  body.rotation.z = Math.sin(freq) * 0.035 * sp; // hip roll adds life
   540→  head.position.y = HEAD_Y + bounce * 0.045 * sp; // head bobs on stride
   541→  head.rotation.x = -0.05 * sp - (sprinting ? 0.08 : 0);
   542→  head.rotation.z = Math.sin(freq) * 0.035 * sp;
   543→  head.rotation.y = 0;
   544→
   545→  if (o.zombie) {
   546→    arms[0].rotation.x = -1.4 + Math.sin(t * 3 + phase) * 0.14;
   547→    arms[1].rotation.x = -1.4 + Math.sin(t * 3 + phase + 1) * 0.14;
   548→    arms[0].rotation.z = 0.1; arms[1].rotation.z = -0.1;
   549→    body.rotation.z = Math.sin(t * 2.4 + phase) * 0.07;
   550→    head.rotation.x = 0.12;
   551→  }
   552→}

   126→   126→/** Shared wind-clock uniform + GPU sway injected into foliage materials. */
   127→   127→const WIND = { value: 0 };
   128→   128→function windify(mat: THREE.Material, amp: number, key: string) {
   129→   129→  mat.onBeforeCompile = (sh) => {
   130→   130→    sh.uniforms.uTime = WIND;
   131→   131→    sh.uniforms.uAmp = { value: amp };
   132→   132→    sh.vertexShader = sh.vertexShader
   133→   133→      .replace(
   134→   134→        "#include <common>",
   135→   135→        "#include <common>\nuniform float uTime;\nuniform float uAmp;",
   136→   136→      )
   137→   137→      .replace(
   138→   138→        "#include <begin_vertex>",
   139→   139→        `#include <begin_vertex>
   140→   140→        {
   141→   141→          float ph = instanceMatrix[3][0] * 0.35 + instanceMatrix[3][2] * 0.27;
   142→   142→          float sw = (sin(uTime * 1.4 + ph) + 0.55 * sin(uTime * 2.4 + ph * 1.7)) * uAmp;
   143→   143→          float lift = clamp((transformed.y + 0.5) * 1.2, 0.0, 1.0);
   144→   144→          transformed.x += sw * lift;
   145→   145→          transformed.z += sw * lift * 0.6;
   146→   146→        }`,
   147→   147→      );
   148→   148→  };
   149→   149→  mat.customProgramCacheKey = () => key;
   150→   150→}

   152→   152→function Forest() {
   153→   153→  const trunks = useRef<THREE.InstancedMesh>(null!);
   154→   154→  const tiers = useRef<THREE.InstancedMesh>(null!);
   155→   155→  const blobs = useRef<THREE.InstancedMesh>(null!);
   156→   156→  const n = useMemo(() => {
   157→   157→    let pine = 0;
   158→   158→    for (const t of WORLD.trees) if (t.kind === 0) pine++;
   159→   159→    return { pine, round: WORLD.trees.length - pine };
   160→   160→  }, []);
   161→   161→
   162→   162→  useLayoutEffect(() => {
   163→   163→    const r = mulberry32(20240);
   164→   164→    // Three stacked cone tiers — base y / radius / height are × tree scale.
   165→   165→    const TIERS = [
   166→   166→      { y: 4.55, rad: 2.6, h: 3.4, shade: 0.93 },
   167→   167→      { y: 6.5, rad: 1.98, h: 2.85, shade: 1.0 },
   168→   168→      { y: 8.15, rad: 1.42, h: 2.35, shade: 1.08 },
   169→   169→    ];
   170→   170→    let ti = 0, bi = 0;
   171→   171→    WORLD.trees.forEach((t, i) => {
   172→   172→      const rot = t.x * 0.7 + t.z * 1.3;
   173→   173→      const hm = range(r, 0.9, 1.14); // per-tree height variety
   174→   174→      tmp.rotation.set(0, rot, 0);
   175→   175→      tmp.scale.set(t.s, t.s, t.s);
   176→   176→      tmp.position.set(t.x, 1.6 * t.s, t.z);
   177→   177→      tmp.updateMatrix();
   178→   178→      trunks.current.setMatrixAt(i, tmp.matrix);
   179→   179→      trunks.current.setColorAt(i, col.set(TRUNK_COLS[i % TRUNK_COLS.length]));
   180→   180→
   181→   181→      if (t.kind === 0) {
   182→   182→        const g = col.set(PINE_COLS[Math.floor(r() * PINE_COLS.length)]).clone();
   183→   183→        for (const tier of TIERS) {
   184→   184→          tmp.rotation.set(0, rot, 0);
   185→   185→          tmp.scale.set(tier.rad * t.s, tier.h * t.s * hm, tier.rad * t.s);
   186→   186→          tmp.position.set(t.x, tier.y * t.s * hm, t.z);
   187→   187→          tmp.updateMatrix();
   188→   188→          tiers.current.setMatrixAt(ti, tmp.matrix);
   189→   189→          tiers.current.setColorAt(ti, col.copy(g).multiplyScalar(tier.shade));
   190→   190→          ti++;
   191→   191→        }
   192→   192→      } else {
   193→   193→        const g = col.set(ROUND_COLS[Math.floor(r() * ROUND_COLS.length)]).clone();
   194→   194→        const put = (dx: number, dy: number, dz: number, rad: number, shade: number) => {
   195→   195→          tmp.rotation.set(0, rot, 0);
   196→   196→          tmp.scale.setScalar(rad * t.s);
   197→   197→          tmp.position.set(t.x + dx * t.s, dy * t.s, t.z + dz * t.s);
   198→   198→          tmp.updateMatrix();
   199→   199→          blobs.current.setMatrixAt(bi, tmp.matrix);
   200→   200→          blobs.current.setColorAt(bi, col.copy(g).multiplyScalar(shade));
   201→   201→          bi++;
   202→   202→        };
   203→   203→        put(0, 5.1, 0, 2.5, 1);        // main puffy crown
   204→   204→        put(0, 6.75, 0, 1.72, 1.09);   // topknot
   205→   205→        for (let k = 0; k < 3; k++) {  // side puffs
   206→   206→          const a = rot + (k * Math.PI * 2) / 3;
   207→   207→          put(Math.cos(a) * 1.6, 4.35, Math.sin(a) * 1.6, 1.58, 0.9);
   208→   208→        }
   209→   209→      }
   210→   210→    });
   211→   211→    tmp.rotation.set(0, 0, 0);
   212→   212→    tmp.scale.setScalar(1);
   213→   213→    trunks.current.count = WORLD.trees.length;
   214→   214→    tiers.current.count = ti;
   215→   215→    blobs.current.count = bi;
   216→   216→    trunks.current.instanceMatrix.needsUpdate = true;
   217→   217→    tiers.current.instanceMatrix.needsUpdate = true;
   218→   218→    blobs.current.instanceMatrix.needsUpdate = true;
   219→   219→    if (trunks.current.instanceColor) trunks.current.instanceColor.needsUpdate = true;
   220→   220→    if (tiers.current.instanceColor) tiers.current.instanceColor.needsUpdate = true;
   221→   221→    if (blobs.current.instanceColor) blobs.current.instanceColor.needsUpdate = true;
   222→   222→  }, [n]);
   223→   223→
   224→   224→  useFrame((s) => { WIND.value = s.clock.elapsedTime; });
   225→   225→
   226→   226→  const trunkMat = useMemo(() => new THREE.MeshLambertMaterial(), []);
   227→   227→  const pineMat = useMemo(() => {
   228→   228→    const m = new THREE.MeshLambertMaterial({ flatShading: true });
   229→   229→    windify(m, 0.05, "wind-pine");
   230→   230→    return m;
   231→   231→  }, []);
   232→   232→  const blobMat = useMemo(() => {
   233→   233→    const m = new THREE.MeshLambertMaterial();
   234→   234→    windify(m, 0.045, "wind-blob");
   235→   235→    return m;
   236→   236→  }, []);
   237→   237→
   238→   238→  return (
   239→   239→    <>
   240→   240→      <instancedMesh ref={trunks} args={[undefined, undefined, WORLD.trees.length]} material={trunkMat}>
   241→   241→        <cylinderGeometry args={[0.34, 0.52, 3.2, 7]} />
   242→   242→      </instancedMesh>
   243→   243→      <instancedMesh ref={tiers} args={[undefined, undefined, Math.max(1, n.pine * 3)]} material={pineMat}>
   244→   244→        <coneGeometry args={[1, 1, 8]} />
   245→   245→      </instancedMesh>
   246→   246→      <instancedMesh ref={blobs} args={[undefined, undefined, Math.max(1, n.round * 5)]} material={blobMat}>
   247→   247→        <sphereGeometry args={[1, 9, 7]} />
   248→   248→      </instancedMesh>
   249→   249→    </>
   250→   250→  );
   251→   251→}

   254→   254→interface WinInst { x: number; y: number; z: number; ry: number }
   255→   255→
   256→   256→function Windows({ clockRef }: { clockRef: React.MutableRefObject<number> }) {
   257→   257→  const glass = useRef<THREE.InstancedMesh>(null!);
   258→   258→  const glow = useRef<THREE.InstancedMesh>(null!);
   259→   259→  const { panes, lit } = useMemo(() => {
   260→   260→    const r = mulberry32(555);
   261→   261→    const panes: WinInst[] = [];
   262→   262→    const lit: WinInst[] = [];
   263→   263→    for (const b of WORLD.buildings) {
   264→   264→      const floors = Math.min(5, Math.max(1, Math.floor((b.h - 3) / 4)));
   265→   265→      const cols = Math.max(1, Math.min(3, Math.floor(b.w * 2 / 5)));
   266→   266→      for (let f = 0; f < floors; f++) {
   267→   267→        const y = 4.2 + f * 4.1;
   268→   268→        if (y > b.h - 2) break;
   269→   269→        for (let c = 0; c < cols; c++) {
   270→   270→          const off = (c - (cols - 1) / 2) * 3.1;
   271→   271→          // front/back faces (±z)
   272→   272→          for (const sd of [-1, 1]) {
   273→   273→            const w: WinInst = { x: b.x + off, y, z: b.z + sd * (b.d + 0.07), ry: sd > 0 ? 0 : Math.PI };
   274→   274→            panes.push(w);
   275→   275→            if (r() < 0.68) lit.push(w);
   276→   276→          }
   277→   277→          // side faces (±x)
   278→   278→          for (const sd of [-1, 1]) {
   279→   279→            const w: WinInst = { x: b.x + sd * (b.w + 0.07), y, z: b.z + off, ry: sd > 0 ? Math.PI / 2 : -Math.PI / 2 };
   280→   280→            panes.push(w);
   281→   281→            if (r() < 0.68) lit.push(w);
   282→   282→          }
   283→   283→        }
   284→   284→      }
   285→   285→    }
   286→   286→    return { panes, lit };
   287→   287→  }, []);
   288→   288→
   289→   289→  useLayoutEffect(() => {
   290→   290→    panes.forEach((w, i) => {
   291→   291→      tmp.position.set(w.x, w.y, w.z);
   292→   292→      tmp.rotation.set(0, w.ry, 0);
   293→   293→      tmp.scale.set(1.15, 1.4, 1);
   294→   294→      tmp.updateMatrix();
   295→   295→      glass.current.setMatrixAt(i, tmp.matrix);
   296→   296→    });
   297→   297→    glass.current.instanceMatrix.needsUpdate = true;
   298→   298→  }, [panes]);
   299→   299→  useLayoutEffect(() => {
   300→   300→    lit.forEach((w, i) => {
   301→   301→      tmp.position.set(w.x, w.y, w.z);
   302→   302→      tmp.rotation.set(0, w.ry, 0);
   303→   303→      tmp.scale.set(1.15, 1.4, 1);
   304→   304→      tmp.updateMatrix();
   305→   305→      glow.current.setMatrixAt(i, tmp.matrix);
   306→   306→    });
   307→   307→    glow.current.count = lit.length;
   308→   308→    glow.current.instanceMatrix.needsUpdate = true;
   309→   309→  }, [lit]);
   310→   310→
   311→   311→  const glassMat = useRef<THREE.MeshLambertMaterial>(null!);
   312→   312→  const glowMat = useRef<THREE.MeshBasicMaterial>(null!);
   313→   313→  const dayC = useMemo(() => new THREE.Color("#5b6a7d"), []);
   314→   314→  const nightC = useMemo(() => new THREE.Color("#1c2230"), []);
   315→   315→  const skyC = useMemo(() => new THREE.Color(), []);
   316→   316→  useFrame(() => {
   317→   317→    const n = nightFactor(clockRef.current);
   318→   318→    if (glassMat.current) glassMat.current.color.copy(skyC.copy(dayC).lerp(nightC, n));
   319→   319→    if (glowMat.current && glow.current) {
   320→   320→      glowMat.current.opacity = Math.max(0, n - 0.35) * 1.5 * (0.82 + Math.sin(performance.now() / 700) * 0.06);
   321→   321→      glow.current.visible = glowMat.current.opacity > 0.02;
   322→   322→    }
   323→   323→  });
   324→   324→
   325→   325→  return (
   326→   326→    <>
   327→   327→      <instancedMesh ref={glass} args={[undefined, undefined, panes.length]}>
   328→   328→        <planeGeometry />
   329→   329→        <meshLambertMaterial ref={glassMat} color="#5b6a7d" />
   330→   330→      </instancedMesh>
   331→   331→      <instancedMesh ref={glow} args={[undefined, undefined, Math.max(1, lit.length)]} frustumCulled={false} visible={false}>
   332→   332→        <planeGeometry />
   333→   333→        <meshBasicMaterial ref={glowMat} color="#ffd76a" transparent opacity={0} depthWrite={false} />
   334→   334→      </instancedMesh>
   335→   335→    </>
   336→   336→  );
   337→   337→}

   417→   417→  const data = useMemo(() => {
   418→   418→    const r = mulberry32(7);
   419→   419→    // Major peaks — ring around the map, each with its own silhouette.
   420→   420→    const majors: Peak[] = Array.from({ length: 26 }, (_, i) => {
   421→   421→      const a = (i / 26) * Math.PI * 2 + range(r, -0.06, 0.06);

   423→   423→      const h = range(r, 85, 178);
   424→   424→      const w = range(r, 52, 96);

   447→   447→          ci: (i + k + 1) % ROCK_COLS.length, tint: range(r, -0.04, 0.05),
   448→   448→          seg: ((i + k + 1) % 3) as 0 | 1 | 2,
   449→   449→          snow: h > 75 ? range(r, 0.58, 0.72) : range(r, 0.82, 0.96),

   457→   457→    });

   460→   460→    return { majors, shoulders, hills, segBody, segCap };
   461→   461→  }, []);
   462→   462→

   471→   471→      tmp.updateMatrix();
   472→   472→      br.setMatrixAt(nb[p.seg], tmp.matrix);
   473→   473→      br.setColorAt(nb[p.seg], col.set(ROCK_COLS[p.ci]).offsetHSL(0, 0, p.tint));
   474→   474→      nb[p.seg]++;
   475→   475→      if (p.snow < 0.98) {

   477→   477→        // Cap is the cone tip region — same triangle, slight overhang so snow drapes.
   478→   478→        tmp.scale.set(p.w * p.sx * (1 - f) * 1.06, p.h * (1 - f) * 1.02, p.w * p.sz * (1 - f) * 1.06);

   480→   480→        tmp.updateMatrix();
   481→   481→        cr.setMatrixAt(nc[p.seg], tmp.matrix);
   482→   482→        nc[p.seg]++;

   486→   486→    data.shoulders.forEach(putPeak);

   488→   488→      tmp.rotation.set(0, hl.x * 0.11, 0);
   489→   489→      tmp.scale.set(hl.w, hl.h, hl.w);
   490→   490→      tmp.position.set(hl.x, hl.h * 0.55, hl.z);
   491→   491→      tmp.updateMatrix();
   492→   492→      hills.current.setMatrixAt(i, tmp.matrix);
   493→   493→      hills.current.setColorAt(i, col.set(HILL_COLS[hl.ci]).offsetHSL(0, 0, hl.tint));
   494→   494→    });
   495→   495→    tmp.rotation.set(0, 0, 0);
   496→   496→    tmp.scale.setScalar(1);

   498→   498→      bodyRefs[s].current.count = nb[s];
   499→   499→      capRefs[s].current.count = nc[s];
   500→   500→      bodyRefs[s].current.instanceMatrix.needsUpdate = true;
   501→   501→      capRefs[s].current.instanceMatrix.needsUpdate = true;
   502→   502→      if (bodyRefs[s].current.instanceColor) bodyRefs[s].current.instanceColor.needsUpdate = true;
   503→   503→    }
   504→   504→    hills.current.instanceMatrix.needsUpdate = true;
   505→   505→    if (hills.current.instanceColor) hills.current.instanceColor.needsUpdate = true;
   506→   506→  }, [data]);
   507→   507→

   510→   510→  const hillMat = useMemo(() => new THREE.MeshLambertMaterial({ flatShading: true }), []);
   511→   511→
   512→   512→  return (
   513→   513→    <>
   514→   514→      <instancedMesh ref={b5} args={[undefined, undefined, Math.max(1, data.segBody[0])]} material={rockMat} frustumCulled={false}>
   515→   515→        <coneGeometry args={[1, 1, 5]} />
   516→   516→      </instancedMesh>
   517→   517→      <instancedMesh ref={b6} args={[undefined, undefined, Math.max(1, data.segBody[1])]} material={rockMat} frustumCulled={false}>
   518→   518→        <coneGeometry args={[1, 1, 6]} />
   519→   519→      </instancedMesh>
   520→   520→      <instancedMesh ref={b7} args={[undefined, undefined, Math.max(1, data.segBody[2])]} material={rockMat} frustumCulled={false}>
   521→   521→        <coneGeometry args={[1, 1, 7]} />
   522→   522→      </instancedMesh>
   523→   523→      <instancedMesh ref={c5} args={[undefined, undefined, Math.max(1, data.segCap[0])]} material={snowMat} frustumCulled={false}>

   620→   620→function Sky({ clockRef }: { clockRef: React.MutableRefObject<number> }) {
   621→   621→  const stars = useRef<THREE.Points>(null!);
   622→   622→  const sun = useRef<THREE.Mesh>(null!);
   623→   623→  const moon = useRef<THREE.Mesh>(null!);

   625→   625→    const r = mulberry32(5);

   628→   628→      const a = range(r, 0, Math.PI * 2), p = range(r, 0.12, 1.35), R = 640;
   629→   629→      pos[i * 3] = Math.cos(a) * Math.cos(p) * R;
   630→   630→      pos[i * 3 + 1] = Math.sin(p) * R * 0.7 + 40;

    17→    17→function Minimap({ engine }: { engine: GameEngine }) {
    18→    18→  const ref = useRef<HTMLCanvasElement>(null);
    19→    19→  const hud = useGame((s) => s.hud);
    20→    20→  useEffect(() => {
    21→    21→    const c = ref.current?.getContext("2d");
    22→    22→    if (!c) return;
    23→    23→    const S = 156, K = S / 700;
    24→    24→    const dot = (x: number, z: number, r: number, col: string, alpha = 1) => {
    25→    25→      c.globalAlpha = alpha;
    26→    26→      c.fillStyle = col;
    27→    27→      c.beginPath();
    28→    28→      c.arc(S / 2 + x * K, S / 2 + z * K, r, 0, 6.3);
    29→    29→      c.fill();
    30→    30→      c.globalAlpha = 1;
    31→    31→    };
    32→    32→    c.clearRect(0, 0, S, S);
    33→    33→    dot(0, 0, 335 * K, "#2d5a2e");
    34→    34→    dot(0, 0, 200 * K, "#7a8f3f");
    35→    35→    dot(0, 0, 130 * K, "#8d8a80");
    36→    36→    dot(-185, 165, 42 * K, "#3a3a46");
    37→    37→    // parliament
    38→    38→    c.fillStyle = "#e0483a";
    39→    39→    c.fillRect(S / 2 - 4, S / 2 - 118 * K - 3, 8, 6);
    40→    40→    // ballot
    41→    41→    c.fillStyle = "#f7c948";
    42→    42→    c.beginPath(); c.arc(S / 2, S / 2 - 100 * K, 2.6, 0, 6.3); c.fill();
    43→    43→    for (const it of engine.items) if (it.active) dot(it.x, it.z, 1.1, "#ffd166", 0.8);
    44→    44→    for (const v of engine.votes) if (v.state === "world") dot(v.x, v.z, 2.2, "#f7c948");
    45→    45→    for (const cop of engine.cops) {
    46→    46→      const chasing = cop.mode === "chase";
    47→    47→      dot(cop.x, cop.z, chasing ? 3 : 2, chasing ? "#ff4d4d" : "#6aa0ff");
    48→    48→    }
    49→    49→    for (const cv of engine.civs) if (cv.mode === "detained") dot(cv.x, cv.z, 3, "#ff8fa3");
    50→    50→    dot(engine.player.x, engine.player.z, 3.6, "#ffffff");
    51→    51→  }, [engine, hud]);
    52→    52→  return (
    53→    53→    <div className="relative origin-top-right scale-[0.72] sm:scale-100">
    54→    54→      <canvas
    55→    55→        ref={ref} width={156} height={156}
    56→    56→        aria-label="Minimap, north at the top"
    57→    57→        className="rounded-xl border-2 border-white/30 shadow-lg"
    58→    58→      />
    59→    59→      <span className="absolute -top-1.5 left-1/2 -translate-x-1/2 rounded-full border border-white/30 bg-black/70 px-1.5 text-[9px] font-black leading-3 text-white">N</span>
    60→    60→    </div>
    61→    61→  );
    62→    62→}

    99→    99→/* ------------- danger compass: arrow toward nearest chasing cop ------------- */
   100→   100→function DangerCompass({ engine }: { engine: GameEngine }) {
   101→   101→  const hud = useGame((s) => s.hud);
   102→   102→  if (!hud.policeNear || hud.detained) return null;
   103→   103→  let best: { d: number; a: number } | null = null;
   104→   104→  for (const cop of engine.cops) {
   105→   105→    if (cop.mode !== "chase") continue;
   106→   106→    const dx = cop.x - engine.player.x, dz = cop.z - engine.player.z;
   107→   107→    const d = Math.hypot(dx, dz);
   108→   108→    if (!best || d < best.d) best = { d, a: Math.atan2(dx, dz) };
   109→   109→  }
   110→   110→  if (!best) return null;
   111→   111→  // 0° = straight ahead of the camera; positive = clockwise (to the right)
   112→   112→  const deg = ((best.a - engine.input.yaw) * 180) / Math.PI;
   113→   113→  return (
   114→   114→    <motion.div
   115→   115→      initial={{ opacity: 0, scale: 0.6 }} animate={{ opacity: 1, scale: 1 }} exit={{ opacity: 0, scale: 0.6 }}
   116→   116→      className="pointer-events-none absolute left-1/2 top-1/2 z-[5]"
   117→   117→      aria-label={`Police ${Math.round(best.d)} meters away`}
   118→   118→    >
   119→   119→      <div style={{ transform: `rotate(${deg}deg)` }}>
   120→   120→        <div
   121→   121→          className="flex flex-col items-center"
   122→   122→          style={{ transform: `translate(-50%, -128px)` }}
   123→   123→        >
   124→   124→          <motion.span animate={{ y: [0, 4, 0] }} transition={{ repeat: Infinity, duration: 0.8 }}>
   125→   125→            <ChevronUp size={26} className="text-rose-400 drop-shadow-[0_1px_4px_rgba(0,0,0,0.8)]" strokeWidth={3.5} />
   126→   126→          </motion.span>
   127→   127→          <span className="rounded-full bg-rose-600/85 px-1.5 text-[9px] font-black leading-4 text-white shadow">
   128→   128→            🚓 {Math.round(best.d)}m
   129→   129→          </span>
   130→   130→        </div>
   131→   131→      </div>
   132→   132→    </motion.div>
   133→   133→  );
   134→   134→}

   136→   136→/* ----------------------------- FPS counter ----------------------------- */
   137→   137→function FpsCounter() {
   138→   138→  const [fps, setFps] = useState(0);
   139→   139→  useEffect(() => {
   140→   140→    let frames = 0, last = performance.now(), raf = 0;
   141→   141→    const tick = () => {
   142→   142→      frames++;
   143→   143→      const now = performance.now();
   144→   144→      if (now - last >= 500) { setFps(Math.round((frames * 1000) / (now - last))); frames = 0; last = now; }
   145→   145→      raf = requestAnimationFrame(tick);
   146→   146→    };
   147→   147→    raf = requestAnimationFrame(tick);
   148→   148→    return () => cancelAnimationFrame(raf);
   149→   149→  }, []);
   150→   150→  return (
   151→   151→    <div aria-label="Frames per second" className="rounded-lg bg-black/40 px-2 py-0.5 text-[10px] font-black tabular-nums text-emerald-300 backdrop-blur">
   152→   152→      {fps} FPS
   153→   153→    </div>
   154→   154→  );
   155→   155→}

   158→   158→function Joystick({ engine }: { engine: GameEngine }) {
   159→   159→  const [knob, setKnob] = useState({ x: 0, y: 0 });
   160→   160→  const ref = useRef<HTMLDivElement>(null);
   161→   161→  const activeId = useRef<number | null>(null);
   162→   162→
   163→   163→  const move = (cx: number, cy: number) => {
   164→   164→    const el = ref.current;
   165→   165→    if (!el) return;
   166→   166→    const r = el.getBoundingClientRect();
   167→   167→    const dx = Math.max(-1, Math.min(1, (cx - (r.left + r.width / 2)) / (r.width / 2)));
   168→   168→    const dy = Math.max(-1, Math.min(1, (cy - (r.top + r.height / 2)) / (r.height / 2)));
   169→   169→    setKnob({ x: dx * 34, y: dy * 34 });
   170→   170→    engine.touchMove.active = true;
   171→   171→    engine.input.f = -dy;
   172→   172→    engine.input.r = dx;
   173→   173→  };
   174→   174→  const end = () => {
   175→   175→    activeId.current = null;
   176→   176→    engine.touchMove.active = false;
   177→   177→    engine.input.f = 0; engine.input.r = 0;
   178→   178→    setKnob({ x: 0, y: 0 });
   179→   179→  };
   180→   180→
   181→   181→  return (
   182→   182→    <div
   183→   183→      ref={ref} data-ui
   184→   184→      className="relative h-32 w-32 rounded-full border-2 border-white/40 bg-white/10 backdrop-blur-sm"
   185→   185→      onTouchStart={(e) => {
   186→   186→        e.preventDefault();
   187→   187→        const t = e.changedTouches[0];
   188→   188→        activeId.current = t.identifier;
   189→   189→        move(t.clientX, t.clientY);
   190→   190→      }}
   191→   191→      onTouchMove={(e) => {
   192→   192→        e.preventDefault();
   193→   193→        for (const t of Array.from(e.changedTouches))
   194→   194→          if (t.identifier === activeId.current) move(t.clientX, t.clientY);
   195→   195→      }}
   196→   196→      onTouchEnd={end}
   197→   197→      onTouchCancel={end}
   198→   198→    >
   199→   199→      <div
   200→   200→        className="absolute left-1/2 top-1/2 h-14 w-14 -translate-x-1/2 -translate-y-1/2 rounded-full border-2 border-white/60 bg-white/30"
   201→   201→        style={{ transform: `translate(calc(-50% + ${knob.x}px), calc(-50% + ${knob.y}px))` }}
   202→   202→      />
   203→   203→    </div>
   204→   204→  );
   205→   205→}

   317→   317→    "Mountain Wall": "text-slate-200",
   318→   318→  };

   326→   326→    setPaused(next);

   336→   336→    };
   337→   337→    addEventListener("keydown", onKey);
   338→   338→    return () => removeEventListener("keydown", onKey);
   339→   339→  }, [victoryOpen]);

   341→   341→  useEffect(() => {
   342→   342→    if (!toasts.length) return;

   369→   369→        <motion.div
   370→   370→          aria-hidden
   371→   371→          className="absolute inset-0"
   372→   372→          animate={{ opacity: [0.35, 0.85, 0.35] }}

   375→   375→        />
   376→   376→      )}
   377→   377→

   379→   379→      <AnimatePresence>{hud.policeNear && <DangerCompass engine={engine} />}</AnimatePresence>

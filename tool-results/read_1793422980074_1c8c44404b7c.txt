    39→/* ------------------------- roads: grid + ring highway + avenue ------------------------- */
    40→function Roads() {
    41→  const SPAN = 250;
    42→  const dashes = useMemo(() => {
    43→    const out: { x: number; z: number; rot: number }[] = [];
    44→    for (const k of ROADS)
    45→      for (let d = -SPAN; d <= SPAN; d += 9) {
    46→        if (ROADS.some((o) => Math.abs(d - o) < 11)) continue; // keep intersections clear
    47→        out.push({ x: k, z: d, rot: 0 });
    48→        out.push({ x: d, z: k, rot: Math.PI / 2 });
    49→      }
    50→    for (let d = AVENUE.z0 + 6; d >= AVENUE.z1 + 5; d -= 8) out.push({ x: AVENUE.x, z: d, rot: 0 });
    51→    return out;
    52→  }, []);
    53→  const ringDashes = useMemo(() => Array.from({ length: 56 }, (_, i) => (i / 56) * Math.PI * 2), []);
    54→
    55→  const ref = useRef<THREE.InstancedMesh>(null!);
    56→  const ringRef = useRef<THREE.InstancedMesh>(null!);
    57→  const lane = useRef<THREE.InstancedMesh>(null!);
    58→  useLayoutEffect(() => {
    59→    dashes.forEach((d, i) => {
    60→      tmp.position.set(d.x, 0.12, d.z);
    61→      tmp.rotation.set(-Math.PI / 2, 0, d.rot);
    62→      tmp.scale.set(0.35, 2.4, 1);
    63→      tmp.updateMatrix();
    64→      ref.current.setMatrixAt(i, tmp.matrix);
    65→    });
    66→    ref.current.instanceMatrix.needsUpdate = true;
    67→    ringDashes.forEach((a, i) => {
    68→      tmp.position.set(Math.cos(a) * RING_R, 0.12, Math.sin(a) * RING_R);
    69→      tmp.rotation.set(-Math.PI / 2, 0, -a);
    70→      tmp.scale.set(0.35, 3.2, 1);
    71→      tmp.updateMatrix();
    72→      ringRef.current.setMatrixAt(i, tmp.matrix);
    73→    });
    74→    ringRef.current.instanceMatrix.needsUpdate = true;
    75→    tmp.rotation.set(0, 0, 0);
    76→    tmp.scale.setScalar(1);
    77→    let i = 0;
    78→    for (const k of ROADS) {
    79→      tmp.rotation.set(-Math.PI / 2, 0, 0);
    80→      tmp.position.set(k, 0.07, 0); tmp.scale.set(ROAD_W, SPAN * 2, 1); tmp.updateMatrix();
    81→      lane.current.setMatrixAt(i++, tmp.matrix);
    82→      tmp.position.set(0, 0.07, k); tmp.scale.set(SPAN * 2, ROAD_W, 1); tmp.updateMatrix();
    83→      lane.current.setMatrixAt(i++, tmp.matrix);
    84→    }
    85→    tmp.position.set(AVENUE.x, 0.07, (AVENUE.z0 + AVENUE.z1) / 2);
    86→    tmp.scale.set(AVENUE.w, AVENUE.z0 - AVENUE.z1, 1); tmp.updateMatrix();
    87→    lane.current.setMatrixAt(i++, tmp.matrix);
    88→    tmp.rotation.set(0, 0, 0); tmp.scale.setScalar(1);
    89→    lane.current.instanceMatrix.needsUpdate = true;
    90→  }, [dashes, ringDashes]);
    91→  return (
    92→    <>
    93→      <instancedMesh ref={lane} args={[undefined, undefined, 11]}>
    94→        <planeGeometry />
    95→        <meshLambertMaterial color="#4a4a52" />
    96→      </instancedMesh>
    97→      <mesh rotation-x={-Math.PI / 2} position-y={0.08}>
    98→        <ringGeometry args={[RING_R - RING_W / 2, RING_R + RING_W / 2, 96]} />
    99→        <meshLambertMaterial color="#45454d" />
   100→      </mesh>
   101→      <instancedMesh ref={ringRef} args={[undefined, undefined, ringDashes.length]}>
   102→        <planeGeometry />
   103→        <meshBasicMaterial color="#f8f4e8" />

   106→        <planeGeometry />
   107→        <meshBasicMaterial color="#f8f4e8" />

   225→/** Shared wind-clock uniform + GPU sway injected into foliage materials. */
   226→const WIND = { value: 0 };
   227→function windify(mat: THREE.Material, amp: number, key: string) {
   228→  mat.onBeforeCompile = (sh) => {
   229→    sh.uniforms.uTime = WIND;
   230→    sh.uniforms.uAmp = { value: amp };
   231→    sh.vertexShader = sh.vertexShader
   232→      .replace(
   233→        "#include <common>",
   234→        "#include <common>\nuniform float uTime;\nuniform float uAmp;",
   235→      )
   236→      .replace(
   237→        "#include <begin_vertex>",
   238→        `#include <begin_vertex>
   239→        {
   240→          float ph = instanceMatrix[3][0] * 0.35 + instanceMatrix[3][2] * 0.27;
   241→          float sw = (sin(uTime * 1.4 + ph) + 0.55 * sin(uTime * 2.4 + ph * 1.7)) * uAmp;
   242→          float lift = clamp((transformed.y + 0.5) * 1.2, 0.0, 1.0);
   243→          transformed.x += sw * lift;
   244→          transformed.z += sw * lift * 0.6;
   245→        }`,
   246→      );
   247→  };
   248→  mat.customProgramCacheKey = () => key;
   249→}

   251→function Forest() {
   252→  const trunks = useRef<THREE.InstancedMesh>(null!);
   253→  const tiers = useRef<THREE.InstancedMesh>(null!);
   254→  const blobs = useRef<THREE.InstancedMesh>(null!);
   255→  const n = useMemo(() => {
   256→    let pine = 0;
   257→    for (const t of WORLD.trees) if (t.kind === 0) pine++;
   258→    return { pine, round: WORLD.trees.length - pine };
   259→  }, []);
   260→
   261→  useLayoutEffect(() => {
   262→    const r = mulberry32(20240);
   263→    // Three stacked cone tiers — base y / radius / height are × tree scale.
   264→    const TIERS = [
   265→      { y: 4.55, rad: 2.6, h: 3.4, shade: 0.93 },
   266→      { y: 6.5, rad: 1.98, h: 2.85, shade: 1.0 },
   267→      { y: 8.15, rad: 1.42, h: 2.35, shade: 1.08 },
   268→    ];
   269→    let ti = 0, bi = 0;
   270→    WORLD.trees.forEach((t, i) => {
   271→      const rot = t.x * 0.7 + t.z * 1.3;
   272→      const hm = range(r, 0.9, 1.14); // per-tree height variety
   273→      tmp.rotation.set(0, rot, 0);
   274→      tmp.scale.set(t.s, t.s, t.s);
   275→      tmp.position.set(t.x, 1.6 * t.s, t.z);
   276→      tmp.updateMatrix();
   277→      trunks.current.setMatrixAt(i, tmp.matrix);
   278→      trunks.current.setColorAt(i, col.set(TRUNK_COLS[i % TRUNK_COLS.length]));
   279→
   280→      if (t.kind === 0) {
   281→        const g = col.set(PINE_COLS[Math.floor(r() * PINE_COLS.length)]).clone();
   282→        for (const tier of TIERS) {
   283→          tmp.rotation.set(0, rot, 0);
   284→          tmp.scale.set(tier.rad * t.s, tier.h * t.s * hm, tier.rad * t.s);
   285→          tmp.position.set(t.x, tier.y * t.s * hm, t.z);
   286→          tmp.updateMatrix();
   287→          tiers.current.setMatrixAt(ti, tmp.matrix);
   288→          tiers.current.setColorAt(ti, col.copy(g).multiplyScalar(tier.shade));
   289→          ti++;
   290→        }
   291→      } else {
   292→        const g = col.set(ROUND_COLS[Math.floor(r() * ROUND_COLS.length)]).clone();
   293→        const put = (dx: number, dy: number, dz: number, rad: number, shade: number) => {
   294→          tmp.rotation.set(0, rot, 0);
   295→          tmp.scale.setScalar(rad * t.s);
   296→          tmp.position.set(t.x + dx * t.s, dy * t.s, t.z + dz * t.s);
   297→          tmp.updateMatrix();
   298→          blobs.current.setMatrixAt(bi, tmp.matrix);
   299→          blobs.current.setColorAt(bi, col.copy(g).multiplyScalar(shade));
   300→          bi++;
   301→        };
   302→        put(0, 5.1, 0, 2.5, 1);        // main puffy crown
   303→        put(0, 6.75, 0, 1.72, 1.09);   // topknot
   304→        for (let k = 0; k < 3; k++) {  // side puffs
   305→          const a = rot + (k * Math.PI * 2) / 3;
   306→          put(Math.cos(a) * 1.6, 4.35, Math.sin(a) * 1.6, 1.58, 0.9);
   307→        }
   308→      }
   309→    });
   310→    tmp.rotation.set(0, 0, 0);
   311→    tmp.scale.setScalar(1);
   312→    trunks.current.count = WORLD.trees.length;
   313→    tiers.current.count = ti;
   314→    blobs.current.count = bi;
   315→    trunks.current.instanceMatrix.needsUpdate = true;
   316→    tiers.current.instanceMatrix.needsUpdate = true;
   317→    blobs.current.instanceMatrix.needsUpdate = true;
   318→    if (trunks.current.instanceColor) trunks.current.instanceColor.needsUpdate = true;
   319→    if (tiers.current.instanceColor) tiers.current.instanceColor.needsUpdate = true;
   320→    if (blobs.current.instanceColor) blobs.current.instanceColor.needsUpdate = true;
   321→  }, [n]);
   322→
   323→  useFrame((s) => { WIND.value = s.clock.elapsedTime; });
   324→
   325→  const trunkMat = useMemo(() => new THREE.MeshLambertMaterial(), []);
   326→  const pineMat = useMemo(() => {
   327→    const m = new THREE.MeshLambertMaterial({ flatShading: true });
   328→    windify(m, 0.05, "wind-pine");
   329→    return m;
   330→  }, []);
   331→  const blobMat = useMemo(() => {
   332→    const m = new THREE.MeshLambertMaterial();
   333→    windify(m, 0.045, "wind-blob");
   334→    return m;
   335→  }, []);
   336→
   337→  return (
   338→    <>
   339→      <instancedMesh ref={trunks} args={[undefined, undefined, WORLD.trees.length]} material={trunkMat}>
   340→        <cylinderGeometry args={[0.34, 0.52, 3.2, 5]} />
   341→      </instancedMesh>
   342→      <instancedMesh ref={tiers} args={[undefined, undefined, Math.max(1, n.pine * 3)]} material={pineMat}>
   343→        <coneGeometry args={[1, 1, 6]} />
   344→      </instancedMesh>
   345→      <instancedMesh ref={blobs} args={[undefined, undefined, Math.max(1, n.round * 5)]} material={blobMat}>
   346→        <sphereGeometry args={[1, 6, 4]} />
   347→      </instancedMesh>
   348→    </>
   349→  );

   355→function Windows({ clockRef }: { clockRef: React.MutableRefObject<number> }) {
   356→  const glass = useRef<THREE.InstancedMesh>(null!);
   357→  const glow = useRef<THREE.InstancedMesh>(null!);
   358→  const { panes, lit } = useMemo(() => {
   359→    const r = mulberry32(555);
   360→    const panes: WinInst[] = [];
   361→    const lit: WinInst[] = [];
   362→    for (const b of WORLD.buildings) {
   363→      if (b.kind === 3) continue; // mall has a glass atrium instead
   364→      let floors: number, y0: number, dy: number, pw: number, ph: number, cols: number, sp: number;
   365→      switch (b.kind) {
   366→        case 0: floors = Math.min(6, Math.floor((b.h - 4) / 4.2)); y0 = 4.4; dy = 4.1; pw = 1.2; ph = 1.5; cols = Math.max(1, Math.min(3, Math.floor(b.w * 2 / 5))); sp = 3.1; break;
   367→        case 1: floors = 1; y0 = 2.4; dy = 0; pw = 1.9; ph = 1.9; cols = 2; sp = 5.0; break;
   368→        case 2: floors = 1; y0 = 3.6; dy = 0; pw = 0.95; ph = 1.05; cols = 2; sp = 3.0; break;
   369→        case 4: floors = 2; y0 = 3.6; dy = 3.0; pw = 1.5; ph = 1.3; cols = 3; sp = 3.2; break;
   370→        case 5: floors = 3; y0 = 3.8; dy = 3.6; pw = 1.3; ph = 1.4; cols = 3; sp = 3.2; break;
   371→        default: floors = 1; y0 = 4.2; dy = 0; pw = 1.1; ph = 1.0; cols = 2; sp = 4.0; break; // barn
   372→      }
   373→      const spread = Math.min(sp, (b.w * 2) / (cols + 0.6));
   374→      for (let f = 0; f < floors; f++) {
   375→        const y = y0 + f * dy;
   376→        if (y > b.h - 1.8) break;
   377→        for (let c = 0; c < cols; c++) {
   378→          const off = (c - (cols - 1) / 2) * spread;
   379→          // front/back faces (±z)
   380→          for (const sd of [-1, 1]) {
   381→            const w: WinInst = { x: b.x + off, y, z: b.z + sd * (b.d + 0.07), ry: sd > 0 ? 0 : Math.PI, w: pw, h: ph };
   382→            panes.push(w);
   383→            if (r() < 0.68) lit.push(w);
   384→          }
   385→          // side faces (±x)
   386→          for (const sd of [-1, 1]) {
   387→            const w: WinInst = { x: b.x + sd * (b.w + 0.07), y, z: b.z + off, ry: sd > 0 ? Math.PI / 2 : -Math.PI / 2, w: pw, h: ph };
   388→            panes.push(w);
   389→            if (r() < 0.68) lit.push(w);
   390→          }
   391→        }
   392→      }
   393→    }
   394→    return { panes, lit };
   395→  }, []);
   396→
   397→  useLayoutEffect(() => {
   398→    panes.forEach((w, i) => {
   399→      tmp.position.set(w.x, w.y, w.z);
   400→      tmp.rotation.set(0, w.ry, 0);
   401→      tmp.scale.set(w.w, w.h, 1);
   402→      tmp.updateMatrix();
   403→      glass.current.setMatrixAt(i, tmp.matrix);
   404→    });
   405→    glass.current.instanceMatrix.needsUpdate = true;
   406→  }, [panes]);
   407→  useLayoutEffect(() => {
   408→    lit.forEach((w, i) => {
   409→      tmp.position.set(w.x, w.y, w.z);
   410→      tmp.rotation.set(0, w.ry, 0);
   411→      tmp.scale.set(w.w, w.h, 1);
   412→      tmp.updateMatrix();
   413→      glow.current.setMatrixAt(i, tmp.matrix);
   414→    });
   415→    glow.current.count = lit.length;
   416→    glow.current.instanceMatrix.needsUpdate = true;
   417→  }, [lit]);
   418→
   419→  const glassMat = useRef<THREE.MeshLambertMaterial>(null!);
   420→  const glowMat = useRef<THREE.MeshBasicMaterial>(null!);
   421→  const dayC = useMemo(() => new THREE.Color("#5b6a7d"), []);
   422→  const nightC = useMemo(() => new THREE.Color("#1c2230"), []);
   423→  const skyC = useMemo(() => new THREE.Color(), []);
   424→  useFrame(() => {
   425→    const n = nightFactor(clockRef.current);
   426→    if (glassMat.current) glassMat.current.color.copy(skyC.copy(dayC).lerp(nightC, n));
   427→    if (glowMat.current && glow.current) {
   428→      glowMat.current.opacity = Math.max(0, n - 0.35) * 1.5 * (0.82 + Math.sin(performance.now() / 700) * 0.06);
   429→      glow.current.visible = glowMat.current.opacity > 0.02;
   430→    }
   431→  });

   448→const CAR_COL_HEX = ["#e0483a", "#2a9d8f", "#f0c040", "#f2f2f2", "#8a5aa8", "#ff8c42", "#4aa3df", "#67b05f"];

   450→function placeCar(
   451→  body: THREE.InstancedMesh, cab: THREE.InstancedMesh, trim: THREE.InstancedMesh,
   452→  wheels: THREE.InstancedMesh | null,
   453→  i: number, x: number, y: number, z: number, rot: number, c: number,
   454→) {
   455→  const fx = Math.sin(rot), fz = Math.cos(rot);
   456→  const cs = Math.cos(rot), sn = Math.sin(rot);
   457→  tmp.rotation.set(0, rot, 0);
   458→  tmp.position.set(x, y + 0.52, z);
   459→  tmp.scale.set(1.9, 1.0, 4.4);
   460→  tmp.updateMatrix();
   461→  body.setMatrixAt(i, tmp.matrix);
   462→  body.setColorAt(i, col.set(CAR_COL_HEX[c % CAR_COL_HEX.length]));
   463→  tmp.position.set(x - fx * 0.4, y + 1.3, z - fz * 0.4);
   464→  tmp.scale.set(1.66, 0.62, 2.05);
   465→  tmp.updateMatrix();
   466→  cab.setMatrixAt(i, tmp.matrix);
   467→  cab.setColorAt(i, col.set("#9fd0e4"));
   468→  tmp.position.set(x, y + 0.16, z);
   469→  tmp.scale.set(2.02, 0.32, 4.55);
   470→  tmp.updateMatrix();
   471→  trim.setMatrixAt(i, tmp.matrix);
   472→  trim.setColorAt(i, col.set("#14181d"));
   473→  if (wheels) {
   474→    for (let k = 0; k < 4; k++) {
   475→      const lx = k % 2 === 0 ? -0.98 : 0.98;
   476→      const lz = k < 2 ? -1.42 : 1.42;
   477→      tmp.rotation.set(0, rot, Math.PI / 2);
   478→      tmp.position.set(x + lx * cs + lz * sn, y + 0.42, z - lx * sn + lz * cs);
   479→      tmp.scale.setScalar(1);
   480→      tmp.updateMatrix();
   481→      wheels.setMatrixAt(i * 4 + k, tmp.matrix);
   482→    }
   483→  }
   484→}

   525→function RingTraffic() {
   526→  const N = 10;
   527→  const data = useMemo(() => Array.from({ length: N }, (_, i) => ({
   528→    r: i % 2 === 0 ? RING_R - 2.7 : RING_R + 2.7,
   529→    dir: i % 2 === 0 ? 1 : -1,
   530→    a: (i / N) * Math.PI * 2,
   531→    sp: 0.05 + (i % 3) * 0.014,
   532→    c: (i * 3 + 1) % CAR_COL_HEX.length,
   533→  })), []);
   534→  const body = useRef<THREE.InstancedMesh>(null!);
   535→  const cab = useRef<THREE.InstancedMesh>(null!);
   536→  const trim = useRef<THREE.InstancedMesh>(null!);

   538→  useFrame((s) => {
   539→    data.forEach((d, i) => {
   540→      const a = d.a + s.clock.elapsedTime * d.sp * d.dir;
   541→      const rot = d.dir > 0 ? -a : Math.PI + a;
   542→      placeCar(body.current, cab.current, trim.current, wheels.current, i, Math.cos(a) * d.r, 0.07, Math.sin(a) * d.r, rot, d.c);
   543→    });
   544→    body.current.instanceMatrix.needsUpdate = true;
   545→    cab.current.instanceMatrix.needsUpdate = true;
   546→    trim.current.instanceMatrix.needsUpdate = true;
   547→    wheels.current.instanceMatrix.needsUpdate = true;
   548→    if (body.current.instanceColor) body.current.instanceColor.needsUpdate = true;
   549→    if (cab.current.instanceColor) cab.current.instanceColor.needsUpdate = true;
   550→    if (trim.current.instanceColor) trim.current.instanceColor.needsUpdate = true;
   551→  });
   552→  return (
   553→    <>
   554→      <instancedMesh ref={body} args={[undefined, undefined, N]} frustumCulled={false}>
   555→        <boxGeometry />
   556→        <meshLambertMaterial />
   557→      </instancedMesh>
   558→      <instancedMesh ref={cab} args={[undefined, undefined, N]} frustumCulled={false}>
   559→        <boxGeometry />
   560→        <meshLambertMaterial />
   561→      </instancedMesh>
   562→      <instancedMesh ref={trim} args={[undefined, undefined, N]} frustumCulled={false}>

  1043→    data.shoulders.forEach(putPeak);
  1044→    data.hills.forEach((hl, i) => {
  1045→      tmp.rotation.set(0, hl.x * 0.11, 0);

  1051→    });
  1052→    tmp.rotation.set(0, 0, 0);

  1056→      capRefs[s].current.count = nc[s];

  1247→/* ------------------------- environment ------------------------- */
  1248→const DAY = new THREE.Color(0x8fd3ff);
  1249→const DUSK = new THREE.Color(0xff9a5a);
  1250→const NIGHT = new THREE.Color(0x0a0e2a);
  1251→
  1252→function Env({ clockRef }: { clockRef: React.MutableRefObject<number> }) {
  1253→  const { scene } = useThree();
  1254→  const hemi = useRef<THREE.HemisphereLight>(null!);
  1255→  const sun = useRef<THREE.DirectionalLight>(null!);
  1256→  const sky = useMemo(() => new THREE.Color(), []);
  1257→  const PI2 = Math.PI * 2;
  1258→  useFrame(() => {
  1259→    const clock = clockRef.current;
  1260→    const n = nightFactor(clock);
  1261→    const h = Math.sin(PI2 * (clock - 0.25));
  1262→    const dk = Math.max(0, 1 - Math.abs(h) * 5);
  1263→    sky.copy(DAY).lerp(NIGHT, n);
  1264→    if (dk > 0) sky.lerp(DUSK, dk * 0.6);
  1265→    scene.background = sky;
  1266→    scene.fog = new THREE.Fog(sky, 90, n > 0.5 ? 300 : 580);
  1267→    hemi.current.intensity = 0.35 + 0.55 * (1 - n);
  1268→    sun.current.intensity = 0.25 + 0.95 * (1 - n);
  1269→    sun.current.position.set(Math.cos(PI2 * (clock - 0.25)) * 200, Math.max(24, h * 200), 90);
  1270→  });
  1271→  return (
  1272→    <>
  1273→      <hemisphereLight ref={hemi} args={[0xffffff, 0x557744, 0.8]} />
  1274→      <directionalLight ref={sun} args={[0xfff2d0, 1]} />
  1275→    </>
  1276→  );
  1277→}

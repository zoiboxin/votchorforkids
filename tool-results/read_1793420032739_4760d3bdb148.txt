    59→    59→/** Collects primitives and merges them into one vertex-coloured mesh. */
    60→    60→class Bucket {
    61→    61→  parts: THREE.BufferGeometry[] = [];
    62→    62→  add(
    63→    63→    geo: THREE.BufferGeometry, color: string, unlit: boolean,
    64→    64→    x = 0, y = 0, z = 0, rx = 0, ry = 0, rz = 0, sx = 1, sy = 1, sz = 1,
    65→    65→  ) {
    66→    66→    const g = geo.clone();
    67→    67→    _m4.compose(_v.set(x, y, z), _q.setFromEuler(_e.set(rx, ry, rz)), _s.set(sx, sy, sz));
    68→    68→    g.applyMatrix4(_m4);
    69→    69→    const c = new THREE.Color(color);
    70→    70→    const n = g.attributes.position.count;
    71→    71→    const arr = new Float32Array(n * 3);
    72→    72→    for (let i = 0; i < n; i++) { arr[i * 3] = c.r; arr[i * 3 + 1] = c.g; arr[i * 3 + 2] = c.b; }
    73→    73→    g.setAttribute("color", new THREE.BufferAttribute(arr, 3));
    74→    74→    for (const k of Object.keys(g.attributes)) if (k !== "position" && k !== "normal" && k !== "uv" && k !== "color") g.deleteAttribute(k);
    75→    75→    this.parts.push(g);
    76→    76→  }
    77→    77→  mesh(kind: "toon" | "basic"): THREE.Mesh | null {
    78→    78→    if (!this.parts.length) return null;
    79→    79→    const merged = mergeGeometries(this.parts, false);
    80→    80→    this.parts.forEach((p) => p.dispose());
    81→    81→    this.parts = [];
    82→    82→    if (!merged) return null;
    83→    83→    merged.clearGroups();
    84→    84→    const mesh = new THREE.Mesh(merged, kind === "toon" ? toonVC() : basicVC());
    85→    85→    mesh.castShadow = false;
    86→    86→    mesh.receiveShadow = false;
    87→    87→    return mesh;
    88→    88→  }
    89→    89→}

   103→   103→function getShadowTex(): THREE.CanvasTexture {
   104→   104→  if (shadowTex) return shadowTex;
   105→   105→  const c = document.createElement("canvas");
   106→   106→  c.width = c.height = 64;
   107→   107→  const ctx = c.getContext("2d")!;
   108→   108→  const g = ctx.createRadialGradient(32, 32, 2, 32, 32, 30);
   109→   109→  g.addColorStop(0, "rgba(0,0,0,0.34)");
   110→   110→  g.addColorStop(1, "rgba(0,0,0,0)");
   111→   111→  ctx.fillStyle = g;
   112→   112→  ctx.fillRect(0, 0, 64, 64);
   113→   113→  shadowTex = new THREE.CanvasTexture(c);
   114→   114→  return shadowTex;

   116→   116→export function blobShadow(r = 0.55): THREE.Mesh {
   117→   117→  const m = new THREE.Mesh(
   118→   118→    G("shadowPlane", () => new THREE.PlaneGeometry(1, 1)),
   119→   119→    new THREE.MeshBasicMaterial({ map: getShadowTex(), transparent: true, depthWrite: false }),
   120→   120→  );
   121→   121→  m.rotation.x = -Math.PI / 2;
   122→   122→  m.scale.setScalar(r * 2);
   123→   123→  m.position.y = 0.11;
   124→   124→  return m;
   125→   125→}

   335→   335→    legB[i].add(sphere(0.145, 9, 7), zombie ? "#3a3a2a" : a.shoe, false, 0, -0.66, 0.06, 0, 0, 0, 1, 0.7, 1.45);
   336→   336→    legB[i].add(sphere(0.1, 8, 6), "#f8f4e8", false, 0, -0.68, 0.19, 0, 0, 0, 0.9, 0.5, 0.7);
   337→   337→  }
   338→   338→  const legs: THREE.Group[] = [];
   339→   339→  for (const i of [0, 1]) {
   340→   340→    const sd = i === 0 ? -1 : 1;
   341→   341→    const leg = new THREE.Group();
   342→   342→    leg.position.set(sd * 0.15, 0.86, 0);
   343→   343→    const m = legB[i].mesh("toon");
   344→   344→    if (m) leg.add(m);
   345→   345→    body.add(leg);
   346→   346→    legs.push(leg);
   347→   347→  }
   348→   348→
   349→   349→  /* bake the merged meshes */
   350→   350→  const bm = bodyToon.mesh("toon"); if (bm) body.add(bm);
   351→   351→  const bu = bodyUnlit.mesh("basic"); if (bu) body.add(bu);
   352→   352→  const hm = headToon.mesh("toon"); if (hm) head.add(hm);
   353→   353→  const hu = headUnlit.mesh("basic"); if (hu) head.add(hu);
   354→   354→
   355→   355→  root.scale.setScalar(AGE_STATS[a.age].scale);
   356→   356→  if (opts.shadow) root.add(blobShadow(0.64));
   357→   357→  const rig: CharRig = { root, body, head, legs, arms, carried: new THREE.Group(), brows, lids };
   358→   358→  root.userData.rig = rig;
   359→   359→  root.userData.look = { ...a };
   360→   360→  root.userData.blink = { next: 1.5 + Math.random() * 3, t: -1, amt: 0 };
   361→   361→  if (zombie) head.rotation.z = 0.16;
   362→   362→  return root;
   363→   363→}

   372→   372→function updateBlink(g: THREE.Object3D, rig: CharRig, t: number) {
   373→   373→  const st = g.userData.blink as { next: number; t: number; amt: number } | undefined;
   374→   374→  if (!st || rig.lids.length < 2) return;
   375→   375→  if (st.t < 0) st.t = t;
   376→   376→  if (t >= st.next) { st.t = t; st.next = t + 1.6 + Math.random() * 3.4; }
   377→   377→  const p = (t - st.t) / 0.16;
   378→   378→  st.amt = p >= 1 ? 0 : Math.sin(Math.min(1, p) * Math.PI);
   379→   379→  const rot = LID_OPEN + (LID_CLOSED - LID_OPEN) * st.amt;
   380→   380→  rig.lids[0].rotation.x = rot;
   381→   381→  rig.lids[1].rotation.x = rot;
   382→   382→}

   384→   384→export function poseCharacter(g: THREE.Object3D, speed: number, t: number, o: PoseOpts = {}) {
   385→   385→  const rig = rigOf(g);
   386→   386→  if (!rig) return;
   387→   387→  const { legs, arms, body, head } = rig;
   388→   388→  const phase = o.phase ?? 0;
   389→   389→
   390→   390→  updateBlink(g, rig, t);
   391→   391→
   392→   392→  if (rig.brows.length === 2) {
   393→   393→    const sad = o.detained ? 0.3 : 0;
   394→   394→    rig.brows[0].rotation.z = -0.08 + sad;
   395→   395→    rig.brows[1].rotation.z = 0.08 - sad;
   396→   396→    const lift = o.detained ? -0.015 : Math.min(0.03, speed * 0.004);
   397→   397→    rig.brows[0].position.y = 0.19 + lift;
   398→   398→    rig.brows[1].position.y = 0.19 + lift;
   399→   399→  }
   400→   400→
   401→   401→  if (o.detained) {
   402→   402→    body.position.y = -0.34 + Math.sin(t * 2.2 + phase) * 0.015;
   403→   403→    body.rotation.x = 0.12;
   404→   404→    legs[0].rotation.x = legs[1].rotation.x = -1.4;
   405→   405→    arms[0].rotation.x = arms[1].rotation.x = 0.3;
   406→   406→    arms[0].rotation.z = 0.15; arms[1].rotation.z = -0.15;
   407→   407→    head.rotation.x = 0.32;
   408→   408→    body.scale.y = 1;
   409→   409→    return;
   410→   410→  }
   411→   411→
   412→   412→  head.rotation.x = 0;
   413→   413→  body.rotation.x = 0;
   414→   414→  body.rotation.z = 0;
   415→   415→
   416→   416→  if (o.airborne) {
   417→   417→    legs[0].rotation.x = -0.75; legs[1].rotation.x = 0.45;
   418→   418→    arms[0].rotation.x = -2.5; arms[1].rotation.x = -2.5;
   419→   419→    arms[0].rotation.z = 0.25; arms[1].rotation.z = -0.25;
   420→   420→    body.rotation.x = -0.1;
   421→   421→    body.position.y = 0;
   422→   422→    body.scale.y = 1.06; body.scale.x = body.scale.z = 0.97;
   423→   423→    return;
   424→   424→  }
   425→   425→
   426→   426→  const sp = Math.min(1.6, speed / 4.5);
   427→   427→  if (sp < 0.06) {
   428→   428→    const b = Math.sin(t * 2 + phase);
   429→   429→    body.position.y = b * 0.02;
   430→   430→    body.scale.y = 1 + b * 0.014;
   431→   431→    body.scale.x = body.scale.z = 1;
   432→   432→    legs[0].rotation.x = legs[1].rotation.x = 0;
   433→   433→    arms[0].rotation.x = b * 0.07; arms[1].rotation.x = -b * 0.07;
   434→   434→    arms[0].rotation.z = 0.07; arms[1].rotation.z = -0.07;
   435→   435→    head.rotation.y = Math.sin(t * 0.6 + phase) * 0.16;
   436→   436→    head.rotation.z = Math.sin(t * 0.9 + phase) * 0.03;
   437→   437→    return;
   438→   438→  }
   439→   439→
   440→   440→  const sprinting = speed > 6.9;
   441→   441→  const freq = t * (5.4 + speed * 1.45) + phase;
   442→   442→  const k = Math.sin(freq) * sp * 0.98;
   443→   443→  legs[0].rotation.x = k;
   444→   444→  legs[1].rotation.x = -k;
   445→   445→  arms[0].rotation.x = -k * 0.9;
   446→   446→  arms[1].rotation.x = k * 0.9;
   447→   447→  arms[0].rotation.z = 0.12;
   448→   448→  arms[1].rotation.z = -0.12;
   449→   449→  const bounce = Math.abs(Math.cos(freq));
   450→   450→  body.position.y = bounce * 0.06 * sp;
   451→   451→  body.scale.y = 1 - bounce * 0.035 * sp;
   452→   452→  body.scale.x = body.scale.z = 1 + bounce * 0.02 * sp;
   453→   453→  body.rotation.x = Math.min(0.3, sp * 0.13) + (sprinting ? 0.1 : 0);
   454→   454→  head.rotation.x = -0.05 * sp - (sprinting ? 0.08 : 0);
   455→   455→  head.rotation.z = Math.sin(freq) * 0.035 * sp;
   456→   456→  head.rotation.y = 0;
   457→   457→
   458→   458→  if (o.zombie) {
   459→   459→    arms[0].rotation.x = -1.4 + Math.sin(t * 3 + phase) * 0.14;
   460→   460→    arms[1].rotation.x = -1.4 + Math.sin(t * 3 + phase + 1) * 0.14;
   461→   461→    arms[0].rotation.z = 0.1; arms[1].rotation.z = -0.1;
   462→   462→    body.rotation.z = Math.sin(t * 2.4 + phase) * 0.07;
   463→   463→    head.rotation.x = 0.12;
   464→   464→  }
   465→   465→}
